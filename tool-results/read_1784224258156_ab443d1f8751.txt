    38→    38→// ─── Helper: decode URL (handles base64 + URL-encoding) ──────────────────────
    39→    39→function decodeInputUrl(raw: string): string {
    40→    40→  let url = raw.trim();
    41→    41→
    42→    42→  // Try base64 decode (common for embed URLs)
    43→    43→  try {
    44→    44→    const decoded = atob(url);
    45→    45→    if (decoded.startsWith('http://') || decoded.startsWith('https://')) {
    46→    46→      return decoded;
    47→    47→    }
    48→    48→  } catch(_e) {
    49→    49→    // Not valid base64, continue
    50→    50→  }
    51→    51→
    52→    52→  // Try URL-decoded base64
    53→    53→  try {
    54→    54→    const decoded = atob(decodeURIComponent(url));
    55→    55→    if (decoded.startsWith('http://') || decoded.startsWith('https://')) {
    56→    56→      return decoded;
    57→    57→    }
    58→    58→  } catch(_e) {
    59→    59→    // Not valid, continue
    60→    60→  }
    61→    61→
    62→    62→  // Try URL decode
    63→    63→  try {
    64→    64→    const decoded = decodeURIComponent(url);
    65→    65→    if (decoded.startsWith('http://') || decoded.startsWith('https://')) {
    66→    66→      return decoded;
    67→    67→    }
    68→    68→  } catch(_e) {
    69→    69→    // Not valid, continue
    70→    70→  }
    71→    71→

    75→    75→// ─── Helper: resolve a potentially relative URL against a base ────────────────
    76→    76→function resolveUrl(target: string, baseUrl: string): string {
    77→    77→  if (target.startsWith('http://') || target.startsWith('https://')) {
    78→    78→    return target;
    79→    79→  }
    80→    80→  if (target.startsWith('//')) {
    81→    81→    return `https:${target}`;
    82→    82→  }
    83→    83→  try {
    84→    84→    const base = new URL(baseUrl);
    85→    85→    if (target.startsWith('/')) {
    86→    86→      return `${base.origin}${target}`;
    87→    87→    }
    88→    88→    // Relative path
    89→    89→    const basePath = base.pathname.substring(0, base.pathname.lastIndexOf('/') + 1);
    90→    90→    return `${base.origin}${basePath}${target}`;
    91→    91→  } catch(_e) {
    92→    92→    return target;
    93→    93→  }
    94→    94→}

   184→   184→interface DecryptCall {
   185→   185→  url: string; // The decrypt endpoint URL

   190→   190→function findDecryptCall(html: string, baseUrl: string): DecryptCall | null {
   191→   191→  // ─── Strategy 1: Standard fetch() with string URL ──────────────────────────
   192→   192→  // Matches: fetch('decrypt.php', { method: 'POST', body: 'stream=XXXX' })
   193→   193→  // Also matches: fetch("decrypt.php", {...}), fetch('./decrypt.php', {...}), etc.
   194→   194→  const fetchPatterns = [
   195→   195→    // fetch with .php endpoint
   196→   196→    /fetch\s*\(\s*["']([^"']*(?:decrypt|token|getlink|resolve|api|get_url|geturl|get_stream)[^"']*\.php[^"']*)["']\s*,\s*\{([^}]*)\}\s*\)/gi,
   197→   197→    // fetch with any endpoint containing 'decrypt'
   198→   198→    /fetch\s*\(\s*["']([^"']*decrypt[^"']*)["']\s*,\s*\{([^}]*)\}\s*\)/gi,
   199→   199→    // fetch with any endpoint containing 'token' or 'getlink'
   200→   200→    /fetch\s*\(\s*["']([^"']*(?:token|getlink|get_stream)[^"']*)["']\s*,\s*\{([^}]*)\}\s*\)/gi,
   201→   201→  ];
   202→   202→
   203→   203→  for (const pattern of fetchPatterns) {
   204→   204→    const match = pattern.exec(html);
   205→   205→    if (match) {
   206→   206→      const fetchUrl = resolveUrl(match[1], baseUrl);
   207→   207→      const optionsStr = match[2];
   208→   208→
   209→   209→      const isPost = /method\s*:\s*["']?POST/i.test(optionsStr);
   210→   210→
   211→   211→      let body = '';
   212→   212→      const bodyMatch = /body\s*:\s*["']([^"']+)["']/i.exec(optionsStr);
   213→   213→      if (bodyMatch) {
   214→   214→        body = bodyMatch[1];
   215→   215→      }
   216→   216→
   217→   217→      let contentType = 'application/x-www-form-urlencoded';
   218→   218→      const ctMatch = /(?:content-type|Content-Type)\s*:\s*["']([^"']+)["']/i.exec(optionsStr);
   219→   219→      if (ctMatch) {
   220→   220→        contentType = ctMatch[1];
   221→   221→      }
   222→   222→
   223→   223→      // If body references a variable (e.g., body: formData), try to resolve it
   224→   224→      if (!body) {
   225→   225→        // Look for FormData .append() patterns
   226→   226→        const appendRegex = /\.append\s*\(\s*["'](\w+)["']\s*,\s*([^)]+)\)/gi;
   227→   227→        const params: string[] = [];
   228→   228→        let appendMatch;
   229→   229→        while ((appendMatch = appendRegex.exec(html)) !== null) {
   230→   230→          const key = appendMatch[1];
   231→   231→          const value = appendMatch[2].trim();
   232→   232→          // If the value is a string literal
   233→   233→          const strMatch = /^["']([^"']+)["']$/.exec(value);
   234→   234→          if (strMatch) {
   235→   235→            params.push(`${key}=${encodeURIComponent(strMatch[1])}`);
   236→   236→          } else {
   237→   237→            // It's a variable reference — try to find its value
   238→   238→            const varValRegex = new RegExp(`(?:var|let|const)\\s+${value.replace(/[.*+?^${}()|[\]\\]/g, '\\$&')}\\s*=\\s*["']([^"']+)["']`, 'i');
   239→   239→            const varValMatch = varValRegex.exec(html);
   240→   240→            if (varValMatch) {
   241→   241→              params.push(`${key}=${encodeURIComponent(varValMatch[1])}`);
   242→   242→            } else {
   243→   243→              // If we can't resolve the variable, use it as-is for the stream ID
   244→   244→              const streamId = extractStreamId(html) || extractStreamIdFromUrl(baseUrl);
   245→   245→              if (streamId && (key === 'stream' || key === 'id' || key === 'code')) {
   246→   246→                params.push(`${key}=${encodeURIComponent(streamId)}`);
   247→   247→              }
   248→   248→            }
   249→   249→          }
   250→   250→        }
   251→   251→        if (params.length > 0) {
   252→   252→          body = params.join('&');
   253→   253→        }
   254→   254→      }
   255→   255→
   256→   256→      // If we still don't have a body but have a stream ID, construct one
   257→   257→      if (!body) {
   258→   258→        const streamId = extractStreamId(html) || extractStreamIdFromUrl(baseUrl);
   259→   259→        if (streamId) {
   260→   260→          body = `stream=${encodeURIComponent(streamId)}`;
   261→   261→        }
   262→   262→      }
   263→   263→
   264→   264→      if (isPost || body) {
   265→   265→        return { url: fetchUrl, body, contentType };
   266→   266→      }
   267→   267→    }
   268→   268→  }
   269→   269→
   270→   270→  // ─── Strategy 2: jQuery $.ajax / $.post patterns ───────────────────────────
   271→   271→  const ajaxPatterns = [
   272→   272→    /\$\.(?:post|ajax)\s*\(\s*["']([^"']*(?:decrypt|token|getlink)[^"']*)["']\s*,\s*\{([^}]*)\}\s*\)/gi,
   273→   273→  ];
   274→   274→
   275→   275→  for (const pattern of ajaxPatterns) {
   276→   276→    const match = pattern.exec(html);
   277→   277→    if (match) {
   278→   278→      const fetchUrl = resolveUrl(match[1], baseUrl);
   279→   279→      const optionsStr = match[2];
   280→   280→
   281→   281→      const streamId = extractStreamId(html) || extractStreamIdFromUrl(baseUrl);
   282→   282→      let body = '';
   283→   283→      const dataMatch = /(?:data|body)\s*:\s*["']([^"']+)["']/i.exec(optionsStr);
   284→   284→      if (dataMatch) {
   285→   285→        body = dataMatch[1];
   286→   286→      } else if (streamId) {
   287→   287→        body = `stream=${encodeURIComponent(streamId)}`;
   288→   288→      }
   289→   289→
   290→   290→      return { url: fetchUrl, body, contentType: 'application/x-www-form-urlencoded' };
   291→   291→    }
   292→   292→  }
   293→   293→
   294→   294→  // ─── Strategy 3: XMLHttpRequest patterns ────────────────────────────────────
   295→   295→  const xhrPatterns = [
   296→   296→    /\.open\s*\(\s*["']POST["']\s*,\s*["']([^"']*(?:decrypt|token|getlink)[^"']*)["']/gi,
   297→   297→  ];
   298→   298→
   299→   299→  for (const pattern of xhrPatterns) {
   300→   300→    const match = pattern.exec(html);
   301→   301→    if (match) {
   302→   302→      const fetchUrl = resolveUrl(match[1], baseUrl);
   303→   303→      const streamId = extractStreamId(html) || extractStreamIdFromUrl(baseUrl);
   304→   304→      const body = streamId ? `stream=${encodeURIComponent(streamId)}` : '';
   305→   305→      return { url: fetchUrl, body, contentType: 'application/x-www-form-urlencoded' };
   306→   306→    }
   307→   307→  }
   308→   308→
   309→   309→  return null;
   310→   310→}

   312→   312→// ─── Helper: try a single POST to a decrypt endpoint and check the response ─
   313→   313→async function tryDecryptPost(
   314→   314→  decryptUrl: string,
   315→   315→  body: string,
   316→   316→  contentType: string,
   317→   317→  referer: string
   318→   318→): Promise<string | null> {
   319→   319→  try {
   320→   320→    const origin = (() => {
   321→   321→      try { return new URL(referer).origin; } catch(_e) { return ''; }
   322→   322→    })();
   323→   323→
   324→   324→    const response = await fetch(decryptUrl, {
   325→   325→      method: 'POST',
   326→   326→      headers: {
   327→   327→        ...COMMON_HEADERS,
   328→   328→        'Content-Type': contentType,
   329→   329→        Referer: referer,
   330→   330→        ...(origin ? { Origin: origin } : {}),
   331→   331→      },
   332→   332→      body,
   333→   333→      signal: AbortSignal.timeout(FETCH_TIMEOUT),
   334→   334→      redirect: 'follow',
   335→   335→    });
   336→   336→
   337→   337→    if (!response.ok) return null;
   338→   338→
   339→   339→    const text = await response.text();
   340→   340→
   341→   341→    // Skip obvious error responses
   342→   342→    if (text.toLowerCase().includes('error') && text.length < 100) {
   343→   343→      console.log(`[Resolve Stream] Decrypt returned error: ${text}`);
   344→   344→      return null;
   345→   345→    }
   346→   346→
   347→   347→    // Check if response contains m3u8 URL
   348→   348→    const m3u8 = findM3u8InText(text) || findM3u8InJS(text);
   349→   349→    if (m3u8) return m3u8;
   350→   350→
   351→   351→    // Try as JSON
   352→   352→    try {
   353→   353→      const json = JSON.parse(text);
   354→   354→      const jsonUrl = json.url || json.link || json.src || json.source || json.stream ||
   355→   355→                      json.data?.url || json.data?.link || json.data?.src;
   356→   356→      if (jsonUrl && typeof jsonUrl === 'string' && M3U8_PATTERNS.some(p => p.test(jsonUrl))) {
   357→   357→        return jsonUrl;
   358→   358→      }
   359→   359→    } catch(_e) {
   360→   360→      // Check if raw text is a URL
   361→   361→      const trimmed = text.trim();
   362→   362→      if (trimmed.startsWith('http') && M3U8_PATTERNS.some(p => p.test(trimmed))) {
   363→   363→        return trimmed;
   364→   364→      }
   365→   365→    }
   366→   366→
   367→   367→    return null;
   368→   368→  } catch(_e) {
   369→   369→    return null;
   370→   370→  }
   371→   371→}

   373→   373→// ─── Helper: try known decrypt endpoints for a domain ────────────────────────

   375→   375→async function tryKnownDecryptEndpoints(
   376→   376→  pageUrl: string,
   377→   377→  streamId: string
   378→   378→): Promise<string | null> {
   379→   379→  try {
   380→   380→    const base = new URL(pageUrl);
   381→   381→    const origin = base.origin;
   382→   382→
   383→   383→    // Common decrypt endpoint paths used by streams.center and similar providers
   384→   384→    const decryptPaths = [
   385→   385→      '/embed/decrypt.php',
   386→   386→      '/embed/player/decrypt.php',
   387→   387→      '/decrypt.php',
   388→   388→      '/api/decrypt.php',
   389→   389→      '/embed/token.php',
   390→   390→      '/token.php',
   391→   391→    ];
   392→   392→
   393→   393→    // Different body formats to try — providers use different parameter names and formats
   394→   394→    const bodyFormats = [
   395→   395→      // URL-encoded form with different param names
   396→   396→      { body: `stream=${encodeURIComponent(streamId)}`, ct: 'application/x-www-form-urlencoded' },
   397→   397→      { body: `id=${encodeURIComponent(streamId)}`, ct: 'application/x-www-form-urlencoded' },
   398→   398→      { body: `code=${encodeURIComponent(streamId)}`, ct: 'application/x-www-form-urlencoded' },
   399→   399→      { body: `channel=${encodeURIComponent(streamId)}`, ct: 'application/x-www-form-urlencoded' },
   400→   400→      { body: `data=${encodeURIComponent(streamId)}`, ct: 'application/x-www-form-urlencoded' },
   401→   401→      // JSON body with different field names
   402→   402→      { body: JSON.stringify({ stream: streamId }), ct: 'application/json' },
   403→   403→      { body: JSON.stringify({ id: streamId }), ct: 'application/json' },
   404→   404→      { body: JSON.stringify({ code: streamId }), ct: 'application/json' },
   405→   405→      { body: JSON.stringify({ channel: streamId }), ct: 'application/json' },
   406→   406→      // Raw stream ID as body
   407→   407→      { body: streamId, ct: 'text/plain' },
   408→   408→    ];
   409→   409→
   410→   410→    for (const path of decryptPaths) {
   411→   411→      const decryptUrl = `${origin}${path}`;
   412→   412→      console.log(`[Resolve Stream] Trying known decrypt endpoint: ${decryptUrl}`);
   413→   413→
   414→   414→      for (const fmt of bodyFormats) {
   415→   415→        const result = await tryDecryptPost(decryptUrl, fmt.body, fmt.ct, pageUrl);
   416→   416→        if (result) {
   417→   417→          console.log(`[Resolve Stream] Decrypt succeeded with body format: ${fmt.body.substring(0, 60)}`);
   418→   418→          return result;
   419→   419→        }
   420→   420→      }
   421→   421→    }
   422→   422→
   423→   423→    return null;
   424→   424→  } catch(_e) {
   425→   425→    return null;
   426→   426→  }
   427→   427→}

   429→   429→// ─── Helper: search for decrypt-related code in minified JS ──────────────────
   430→   430→// Scans large JS blobs for any reference to decrypt/token endpoints
   431→   431→function findDecryptInMinifiedJS(jsContent: string, baseUrl: string): DecryptCall | null {
   432→   432→  // Look for any string containing 'decrypt' (minified JS may use variable names or concatenation)
   433→   433→  const decryptStringRegex = /"([^"]*decrypt[^"]*)"|'([^']*decrypt[^']*)'/gi;
   434→   434→  let match;
   435→   435→  while ((match = decryptStringRegex.exec(jsContent)) !== null) {
   436→   436→    const foundStr = match[1] || match[2];
   437→   437→    if (foundStr && (foundStr.endsWith('.php') || foundStr.includes('.php?') || foundStr.startsWith('/') || foundStr.startsWith('http'))) {
   438→   438→      const fetchUrl = resolveUrl(foundStr, baseUrl);
   439→   439→      const streamId = extractStreamId(jsContent) || extractStreamIdFromUrl(baseUrl);
   440→   440→      const body = streamId ? `stream=${encodeURIComponent(streamId)}` : '';
   441→   441→      console.log(`[Resolve Stream] Found decrypt string in minified JS: ${foundStr} -> ${fetchUrl}`);
   442→   442→      return { url: fetchUrl, body, contentType: 'application/x-www-form-urlencoded' };
   443→   443→    }
   444→   444→  }
   445→   445→
   446→   446→  // Also look for URL patterns near the word 'decrypt'
   447→   447→  const nearDecryptRegex = /decrypt[^\n]{0,200}?(https?:\/\/[^\s"'<>]+|\/[\w\/]+\.php[^\s"'<>]*)/gi;
   448→   448→  while ((match = nearDecryptRegex.exec(jsContent)) !== null) {
   449→   449→    const foundUrl = match[1];
   450→   450→    if (foundUrl) {
   451→   451→      const fetchUrl = resolveUrl(foundUrl, baseUrl);
   452→   452→      const streamId = extractStreamId(jsContent) || extractStreamIdFromUrl(baseUrl);
   453→   453→      const body = streamId ? `stream=${encodeURIComponent(streamId)}` : '';
   454→   454→      console.log(`[Resolve Stream] Found URL near 'decrypt' in minified JS: ${foundUrl} -> ${fetchUrl}`);
   455→   455→      return { url: fetchUrl, body, contentType: 'application/x-www-form-urlencoded' };
   456→   456→    }
   457→   457→  }
   458→   458→
   459→   459→  return null;
   460→   460→}

   523→   523→// ─── Core: resolve the embed chain ───────────────────────────────────────────
   524→   524→interface ResolveResult {
   525→   525→  resolvedUrl: string;
   526→   526→  originalUrl: string;
   527→   527→  type: 'hls' | 'iframe';
   528→   528→  resolved: boolean;
   529→   529→}
   530→   530→
   531→   531→async function resolveEmbedChain(embedUrl: string): Promise<ResolveResult> {
   532→   532→  const result: ResolveResult = {
   533→   533→    resolvedUrl: embedUrl,
   534→   534→    originalUrl: embedUrl,
   535→   535→    type: 'iframe',
   536→   536→    resolved: false,
   537→   537→  };
   538→   538→
   539→   539→  // Check if the URL is already an m3u8 URL
   540→   540→  if (M3U8_PATTERNS.some(p => p.test(embedUrl))) {
   541→   541→    console.log(`[Resolve Stream] URL is already m3u8: ${embedUrl}`);
   542→   542→    return { resolvedUrl: embedUrl, originalUrl: embedUrl, type: 'hls', resolved: true };
   543→   543→  }
   544→   544→
   545→   545→  // Step 1: Fetch the initial embed page
   546→   546→  const html = await fetchHtml(embedUrl);
   547→   547→  if (!html) {
   548→   548→    console.warn(`[Resolve Stream] Could not fetch embed page: ${embedUrl}`);
   549→   549→    return result;
   550→   550→  }
   551→   551→
   552→   552→  // Step 2: Check for direct m3u8 URL in the initial page
   553→   553→  const directM3u8 = findM3u8InText(html) || findM3u8InJS(html);
   554→   554→  if (directM3u8) {
   555→   555→    console.log(`[Resolve Stream] Found direct m3u8 in embed page: ${directM3u8}`);
   556→   556→    return { resolvedUrl: directM3u8, originalUrl: embedUrl, type: 'hls', resolved: true };
   557→   557→  }
   558→   558→
   559→   559→  // Step 3: Look for iframe src URLs and follow them
   560→   560→  const iframeSrcs = extractIframeSrcs(html, embedUrl);
   561→   561→  console.log(`[Resolve Stream] Found ${iframeSrcs.length} iframe(s): ${iframeSrcs.map(s => s.substring(0, 80)).join(', ')}`);
   562→   562→
   563→   563→  // Track visited URLs to prevent loops
   564→   564→  const visited = new Set<string>([embedUrl]);
   565→   565→
   566→   566→  // BFS through iframe chain up to MAX_CHAIN_DEPTH
   567→   567→  const queue: Array<{ url: string; depth: number; parentHtml: string }> = iframeSrcs
   568→   568→    .filter(src => !visited.has(src))
   569→   569→    .map(src => {
   570→   570→      visited.add(src);
   571→   571→      return { url: src, depth: 1, parentHtml: html };
   572→   572→    });
   573→   573→
   574→   574→  while (queue.length > 0) {
   575→   575→    const { url: currentUrl, depth, parentHtml } = queue.shift()!;
   576→   576→
   577→   577→    if (depth > MAX_CHAIN_DEPTH) {
   578→   578→      console.warn(`[Resolve Stream] Max chain depth reached at: ${currentUrl}`);
   579→   579→      continue;
   580→   580→    }
   581→   581→
   582→   582→    // Fetch the iframe page
   583→   583→    const iframeHtml = await fetchHtml(currentUrl, embedUrl);
   584→   584→    if (!iframeHtml) continue;
   585→   585→
   586→   586→    // Check for direct m3u8 in the iframe page
   587→   587→    const iframeM3u8 = findM3u8InText(iframeHtml) || findM3u8InJS(iframeHtml);
   588→   588→    if (iframeM3u8) {
   589→   589→      console.log(`[Resolve Stream] Found m3u8 in iframe page (depth ${depth}): ${iframeM3u8}`);
   590→   590→      return { resolvedUrl: iframeM3u8, originalUrl: embedUrl, type: 'hls', resolved: true };
   591→   591→    }
   592→   592→
   593→   593→    // Look for decrypt.php call pattern in the JS
   594→   594→    let decryptCall = findDecryptCall(iframeHtml, currentUrl);
   595→   595→
   596→   596→    // If not found with standard patterns, try scanning minified JS
   597→   597→    if (!decryptCall) {
   598→   598→      decryptCall = findDecryptInMinifiedJS(iframeHtml, currentUrl);
   599→   599→    }
   600→   600→
   601→   601→    if (decryptCall) {
   602→   602→      console.log(`[Resolve Stream] Found decrypt call at depth ${depth}: ${decryptCall.url}`);
   603→   603→
   604→   604→      const decryptResponse = await postDecrypt(
   605→   605→        decryptCall.url,
   606→   606→        decryptCall.body,
   607→   607→        decryptCall.contentType,
   608→   608→        currentUrl
   609→   609→      );
   610→   610→
   611→   611→      if (decryptResponse) {
   612→   612→        const decryptM3u8 = findM3u8InText(decryptResponse) || findM3u8InJS(decryptResponse);
   613→   613→        if (decryptM3u8) {
   614→   614→          console.log(`[Resolve Stream] Got m3u8 from decrypt: ${decryptM3u8}`);
   615→   615→          return { resolvedUrl: decryptM3u8, originalUrl: embedUrl, type: 'hls', resolved: true };
   616→   616→        }
   617→   617→
   618→   618→        // Try parsing as JSON
   619→   619→        try {
   620→   620→          const json = JSON.parse(decryptResponse);
   621→   621→          const jsonUrl = json.url || json.link || json.src || json.source || json.stream ||
   622→   622→                          json.data?.url || json.data?.link || json.data?.src;
   623→   623→          if (jsonUrl && typeof jsonUrl === 'string') {
   624→   624→            if (M3U8_PATTERNS.some(p => p.test(jsonUrl))) {
   625→   625→              console.log(`[Resolve Stream] Got m3u8 from decrypt JSON: ${jsonUrl}`);
   626→   626→              return { resolvedUrl: jsonUrl, originalUrl: embedUrl, type: 'hls', resolved: true };
   627→   627→            }
   628→   628→            // If it's another URL, follow it
   629→   629→            if (jsonUrl.startsWith('http')) {
   630→   630→              console.log(`[Resolve Stream] Decrypt returned another URL: ${jsonUrl}`);
   631→   631→              if (!visited.has(jsonUrl)) {
   632→   632→                visited.add(jsonUrl);
   633→   633→                queue.push({ url: jsonUrl, depth: depth + 1, parentHtml: iframeHtml });
   634→   634→              }
   635→   635→            }
   636→   636→          }
   637→   637→        } catch(_e) {
   638→   638→          // Not JSON, check if the raw text is a URL
   639→   639→          const trimmed = decryptResponse.trim();
   640→   640→          if (trimmed.startsWith('http') && M3U8_PATTERNS.some(p => p.test(trimmed))) {
   641→   641→            console.log(`[Resolve Stream] Decrypt returned raw m3u8: ${trimmed}`);
   642→   642→            return { resolvedUrl: trimmed, originalUrl: embedUrl, type: 'hls', resolved: true };
   643→   643→          }
   644→   644→        }
   645→   645→      }
   646→   646→    }
   647→   647→
   648→   648→    // ── Strategy: If on a known decrypt-chain domain, try known endpoints directly ──
   649→   649→    // The JS may be too minified to parse, so we try brute-force common paths
   650→   650→    const isDecryptChainDomain = DECRYPT_CHAIN_DOMAINS.some(d => currentUrl.includes(d));
   651→   651→    if (isDecryptChainDomain) {
   652→   652→      const streamId = extractStreamId(iframeHtml) || extractStreamIdFromUrl(currentUrl);
   653→   653→      if (streamId) {
   654→   654→        console.log(`[Resolve Stream] On decrypt-chain domain, trying known endpoints with stream ID: ${streamId}`);
   655→   655→        const knownM3u8 = await tryKnownDecryptEndpoints(currentUrl, streamId);
   656→   656→        if (knownM3u8) {
   657→   657→          console.log(`[Resolve Stream] Got m3u8 from known endpoint: ${knownM3u8}`);
   658→   658→          return { resolvedUrl: knownM3u8, originalUrl: embedUrl, type: 'hls', resolved: true };
   659→   659→        }
   660→   660→      }
   661→   661→    }
   662→   662→
   663→   663→    // Look for more iframes within this iframe page
   664→   664→    const nestedIframes = extractIframeSrcs(iframeHtml, currentUrl);
   665→   665→    for (const nestedSrc of nestedIframes) {
   666→   666→      if (!visited.has(nestedSrc)) {
   667→   667→        visited.add(nestedSrc);
   668→   668→        queue.push({ url: nestedSrc, depth: depth + 1, parentHtml: iframeHtml });
   669→   669→      }
   670→   670→    }
   671→   671→  }
   672→   672→
   673→   673→  // Step 4: Handle special cases — fltvhd.com / tvtvhd.com / tarjetaroja chain
   674→   674→  // These sites embed the m3u8 URL directly in their page source
   675→   675→  if (embedUrl.includes('fltvhd.com') || embedUrl.includes('tvtvhd.com') || embedUrl.includes('tarjetaroja') || embedUrl.includes('rojadirecta')) {
   676→   676→    console.log(`[Resolve Stream] Trying tarjetaroja/fltvhd-style extraction for: ${embedUrl}`);
   677→   677→    const scriptRegex = /<script[^>]*>([\s\S]*?)<\/script>/gi;
   678→   678→    let scriptMatch;
   679→   679→    while ((scriptMatch = scriptRegex.exec(html)) !== null) {
   680→   680→      const scriptContent = scriptMatch[1];
   681→   681→      const scriptM3u8 = findM3u8InJS(scriptContent) || findM3u8InText(scriptContent);
   682→   682→      if (scriptM3u8) {
   683→   683→        console.log(`[Resolve Stream] Found m3u8 in script tag: ${scriptM3u8}`);
   684→   684→        return { resolvedUrl: scriptM3u8, originalUrl: embedUrl, type: 'hls', resolved: true };
   685→   685→      }
   686→   686→    }
   687→   687→
   688→   688→    // Also try looking for Clappr or other player configurations
   689→   689→    const playerConfigRegex = /(?:new\s+Clappr\.Player|Hls\.js|videojs|jwplayer)\s*\(([\s\S]*?)\)\s*;/gi;
   690→   690→    let playerMatch;
   691→   691→    while ((playerMatch = playerConfigRegex.exec(html)) !== null) {
   692→   692→      const configStr = playerMatch[1];
   693→   693→      const configM3u8 = findM3u8InText(configStr) || findM3u8InJS(configStr);
   694→   694→      if (configM3u8) {
   695→   695→        console.log(`[Resolve Stream] Found m3u8 in player config: ${configM3u8}`);
   696→   696→        return { resolvedUrl: configM3u8, originalUrl: embedUrl, type: 'hls', resolved: true };
   697→   697→      }
   698→   698→    }
   699→   699→  }
   700→   700→
   701→   701→  // Step 5: Last resort — scan ALL script tags for m3u8 patterns
   702→   702→  const allScriptRegex = /<script[^>]*>([\s\S]*?)<\/script>/gi;
   703→   703→  let scriptMatch;
   704→   704→  while ((scriptMatch = allScriptRegex.exec(html)) !== null) {
   705→   705→    const scriptContent = scriptMatch[1];
   706→   706→    const jsM3u8 = findM3u8InJS(scriptContent) || findM3u8InText(scriptContent);
   707→   707→    if (jsM3u8) {
   708→   708→      console.log(`[Resolve Stream] Found m3u8 in script tag (last resort): ${jsM3u8}`);
   709→   709→      return { resolvedUrl: jsM3u8, originalUrl: embedUrl, type: 'hls', resolved: true };
   710→   710→    }
   711→   711→  }
   712→   712→
   713→   713→  // Step 6: If we found iframe URLs but no m3u8, return the deepest iframe we found
   714→   714→  if (iframeSrcs.length > 0) {
   715→   715→    // Return the first iframe URL that looks like a player page
   716→   716→    const playerIframe = iframeSrcs.find(
   717→   717→      src => src.includes('hls.php') || src.includes('player') || src.includes('embed') || src.includes('stream')
   718→   718→    );
   719→   719→    if (playerIframe) {
   720→   720→      console.log(`[Resolve Stream] Returning iframe URL (fallback): ${playerIframe}`);
   721→   721→      return { resolvedUrl: playerIframe, originalUrl: embedUrl, type: 'iframe', resolved: false };
   722→   722→    }
   723→   723→  }
   724→   724→
   725→   725→  console.warn(`[Resolve Stream] Could not resolve m3u8 for: ${embedUrl}`);
   726→   726→  return result;
   727→   727→}
