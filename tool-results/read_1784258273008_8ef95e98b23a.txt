   402→export default function Home() {
   403→  const { currentView, fetchChannels, fetchFootballMatches, fetchBasketballMatches, fetchHesgoalMatches, mergeHesgoalStreams, language, theme, setTheme, setLanguage, footballMatches } = useAppStore();
   404→  const { isOnline } = usePWA();
   405→  const [footerPrivacyOpen, setFooterPrivacyOpen] = useState(false);
   406→
   407→  // Initialize theme and language from localStorage on mount (after hydration)
   408→  useEffect(() => {
   409→    if (typeof window !== 'undefined') {
   410→      // Theme
   411→      const savedTheme = localStorage.getItem('goalstream_theme');
   412→      if (savedTheme === 'light' || savedTheme === 'dark') {
   413→        setTheme(savedTheme);
   414→      } else {
   415→        document.documentElement.classList.add('dark');
   416→      }
   417→      // Language
   418→      const savedLang = localStorage.getItem('goalstream_language');
   419→      if (savedLang && ['en', 'fr', 'ar', 'es', 'pt'].includes(savedLang)) {
   420→        setLanguage(savedLang);
   421→      }
   422→    }
   423→  }, [setTheme, setLanguage]);
   424→
   425→  useEffect(() => {
   426→    // Initial fetch: today's matches first (fast), then near future in background
   427→    const timer = setTimeout(() => {
   428→      const now = new Date();
   429→      const d = (offset: number) => {
   430→        const dt = new Date(now.getTime() + offset * 24 * 60 * 60 * 1000);
   431→        return `${dt.getFullYear()}${String(dt.getMonth() + 1).padStart(2, '0')}${String(dt.getDate()).padStart(2, '0')}`;
   432→      };
   433→      // Fetch today + past 3 days first
   434→      fetchFootballMatches([d(-3), d(-2), d(-1), d(0)]);
   435→      // Fetch next 7 days after a delay to avoid OOM
   436→      setTimeout(() => {
   437→        const dates: string[] = [];
   438→        for (let i = 1; i <= 7; i++) dates.push(d(i));
   439→        fetchFootballMatches(dates);
   440→      }, 5000);
