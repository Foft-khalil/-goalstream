    26→async function checkStreamsBatchLight(urls: string[], concurrency: number = 4, timeoutMs: number = 4000): Promise<Map<string, boolean>> {
    27→  const results = new Map<string, boolean>();
    28→  for (let i = 0; i < urls.length; i += concurrency) {
    29→    const batch = urls.slice(i, i + concurrency);
    30→    const batchResults = await Promise.allSettled(batch.map(async (url) => {
    31→      try {
    32→        const ctrl = new AbortController();
    33→        const t = setTimeout(() => ctrl.abort(), timeoutMs);
    34→        const isHls = url.includes('.m3u8') || url.includes('m3u8');
    35→        const res = await fetch(url, {
    36→          method: isHls ? 'GET' : 'HEAD',
    37→          signal: ctrl.signal,
    38→          headers: isHls ? { 'Range': 'bytes=0-1' } : {},
    39→          redirect: 'follow',
    40→        });
    41→        clearTimeout(t);
    42→        return { url, healthy: res.ok || res.status === 206 || res.status === 302 };
    43→      } catch(_e) { return { url, healthy: false }; }
    44→    }));
    45→    for (const r of batchResults) {
    46→      if (r.status === 'fulfilled') results.set(r.value.url, r.value.healthy);
    47→      else results.set(batch[batchResults.indexOf(r)], false);
    48→    }
    49→  }
    50→  return results;
    51→}

   339→  const a = awayTeam.toLowerCase();

   344→      countries.add(country);
   345→    }

   347→
   348→  // Always include France as a default for French-speaking users
   349→  countries.add('fr');
   350→
   351→  return Array.from(countries);
   352→}

   354→/**
   355→ * Use web search to find the actual broadcaster for a specific match.
   356→ */
   357→async function findBroadcasterViaSearch(
   358→  homeTeam: string,
   359→  awayTeam: string,
   360→  competition: string,
   361→  sport: string = 'football'
   362→): Promise<string[]> {
   363→  try {
   364→    const ZAI = (await import('z-ai-web-dev-sdk')).default;
   365→    const sdk = await ZAI.create();
   366→    const sportLabel = sport === 'basketball' ? 'basketball' : 'football';
   367→
   368→    // Search in French for French broadcasting context
   369→    const results = await sdk.functions.invoke('web_search', {
   370→      query: `${homeTeam} vs ${awayTeam} ${competition} ${sportLabel} chaine TV diffusion direct streaming 2025`,
   371→      num: 5,
   372→      recency_days: 7,
   373→    });
   374→
   375→    if (!results || results.length === 0) return [];
   376→
   377→    const snippets = results.map((r: any) => r.snippet).join('\n');
   378→
   379→    const chatResponse = await sdk.chat.completions.create({
   380→      messages: [
   381→        {
   382→          role: 'system',
   383→          content: `Tu es un expert en diffusion sportive française et internationale. À partir des résultats de recherche, identifie les chaînes de TV qui diffusent ce match. Retourne UNIQUEMENT les noms des chaînes séparés par des virgules (ex: "Canal+, beIN Sports, RMC Sport"). Si tu ne peux pas déterminer, retourne "Inconnu". N'ajoute aucune explication.`,
   384→        },
   385→        {
   386→          role: 'user',
   387→          content: `Match: ${homeTeam} vs ${awayTeam}\nCompétition: ${competition}\nSport: ${sportLabel}\n\nRésultats de recherche:\n${snippets}`,
   388→        },
   389→      ],
   390→    });
   391→
   392→    const response = chatResponse?.choices?.[0]?.message?.content?.trim();
   393→    if (!response || response === 'Inconnu' || response === 'Unknown') return [];
   394→
   395→    return response.split(',').map((s: string) => s.trim()).filter((s: string) => s.length > 1);
   396→  } catch (err) {
   397→    console.warn('[Match Stream API] Web search for broadcaster failed:', err);
   398→    return [];
   399→  }
   400→}

   405→function getKnownBroadcasters(competition: string): Array<{ broadcaster: string; countries: string[]; priority: number }> {

   407→  for (const [key, broadcasters] of Object.entries(COMPETITION_BROADCASTERS)) {
   408→    if (compLower.includes(key)) return broadcasters;
   409→  }
   410→  return [];
   411→}

   413→/**
   414→ * Find IPTV channels that match broadcaster names.
   415→ * Now includes country-aware priority boosting.
   416→ */
   417→function matchBroadcasterToIPTV(
   418→  broadcasters: Array<{ broadcaster: string; countries: string[]; priority: number }>,
   419→  iptvChannels: Array<{ name: string; url: string; logo: string; group: string; country: string; source: string }>,
   420→  teamCountries: string[],
   421→  sport: string = 'football'
   422→): Array<{ name: string; url: string; logo: string; group: string; country: string; relevance: number; matchedBroadcaster: string; health?: string }> {
   423→  const results: Array<{ name: string; url: string; logo: string; group: string; country: string; relevance: number; matchedBroadcaster: string; health?: string }> = [];
   424→
   425→  // Blacklist: channel names that should never appear in sports results
   426→  const sportsBlacklist = [
   427→    'combat', 'strongman', 'freesports', 'poker', 'casino', 'lottery',
   428→    'racing.com', 'horse', 'dog racing', 'wrestling', 'boxing',
   429→    'fitness', 'gym', 'workout', 'yoga', 'outdoors', 'hunting', 'fishing',
   430→  ];
   431→
   432→  for (const bcast of broadcasters) {
   433→    const bLower = bcast.broadcaster.toLowerCase();
   434→    // Find IPTV keywords for this broadcaster
   435→    let keywords: string[] = [bLower];
   436→    for (const [bcastKey, kwList] of Object.entries(BROADCASTER_TO_IPTV)) {
   437→      if (bLower.includes(bcastKey) || bcastKey.includes(bLower)) {
   438→        keywords = kwList;
   439→        break;
   440→      }
   441→    }
   442→
   443→    // Find matching IPTV channels
   444→    for (const ch of iptvChannels) {
   445→      const nameLower = ch.name.toLowerCase();
   446→      const groupLower = (ch.group || '').toLowerCase();
   447→      const isMatch = keywords.some(kw => nameLower.includes(kw) || groupLower.includes(kw));
   448→
   449→      if (isMatch) {
   450→        // Skip blacklisted channel types
   451→        const isBlacklisted = sportsBlacklist.some(bl => nameLower.includes(bl));
   452→        if (isBlacklisted) continue;
   453→
   454→        // Skip DAZN Combat (not football/basketball)
   455→        if (nameLower.includes('dazn') && nameLower.includes('combat')) continue;
   456→
   457→        // Skip channels with "Canal" but not "Canal+" or "C+" (e.g., "Canal 32" is not Canal+)
   458→        if (keywords.some(kw => kw.includes('canal+') || kw.includes('c+') || kw.includes('canal sport') || kw.includes('canal foot'))) {
   459→          if (nameLower.includes('canal') &&
   460→              !nameLower.includes('canal+') && !nameLower.includes('c+') &&
   461→              !nameLower.includes('canal plus') && !nameLower.includes('canal sport') &&
   462→              !nameLower.includes('canal foot') && !nameLower.includes('canal liga')) {
   463→            continue; // "Canal 32", "Canal 6", etc. are NOT Canal+
   464→          }
   465→          // Also reject channels matching the pattern "Canal <number>" (e.g., Canal 6, Canal 7, Canal 10, Canal 13, Canal 26)
   466→          if (/\bcanal\s+\d+/i.test(ch.name)) {
   467→            continue;
   468→          }
   469→        }
   470→
   471→        let relevance = bcast.priority * 10; // Base score from competition priority
   472→
   473→        // HLS streams get a big bonus (they actually work with our player)
   474→        if (ch.url.includes('.m3u8') || ch.url.includes('m3u8')) relevance += 15;
   475→        // HTTPS bonus
   476→        if (ch.url.startsWith('https')) relevance += 5;
   477→
   478→        // Country match bonus — if the channel's source matches a team country
   479→        const channelSource = (ch.source || '').toLowerCase();
   480→        const channelCountry = (ch.country || '').toLowerCase();
   481→        for (const teamCountry of teamCountries) {
   482→          if (channelSource.includes(teamCountry) || channelCountry === teamCountry) {
   483→            relevance += 20; // Big bonus for country-matched channels
   484→          }
   485→        }
   486→
   487→        // If the broadcaster's countries match the team countries, extra bonus
   488→        for (const bcastCountry of bcast.countries) {
   489→          if (teamCountries.includes(bcastCountry)) {
   490→            relevance += 10;
   491→            break;
   492→          }
   493→        }
   494→
   495→        // Check health status from the health map
   496→        const healthStatus = getChannelHealth(ch.url);
   497→        if (healthStatus === 'online') relevance += 25;
   498→        if (healthStatus === 'offline') relevance -= 30; // Heavy penalty for known-dead channels
   499→
   500→        // Avoid duplicates
   501→        if (!results.find(r => r.url === ch.url)) {
   502→          results.push({ ...ch, relevance, matchedBroadcaster: bcast.broadcaster, health: healthStatus });
   503→        }
   504→      }
   505→    }
   506→  }
   507→
   508→  return results.sort((a, b) => b.relevance - a.relevance);
   509→}

   903→    }).slice(0, 12);
   904→
   905→    // Add "Hors ligne" label to offline channels
   906→    for (const ch of finalChannels) {
   907→      if (ch.health === 'offline') {
   908→        ch.name = ch.name + ' ⛔ Hors ligne';
   909→      }
   910→    }
   911→
   912→    // If no specific match found, return country-specific sports channels
   913→    if (finalChannels.length === 0) {
   914→      const generalSports = allChannels
   915→        .filter((ch) => {
   916→          const nameLower = ch.name.toLowerCase();
   917→          const groupLower = (ch.group || '').toLowerCase();
   918→          return nameLower.includes('sport') || groupLower.includes('sport');
   919→        })
   920→        .slice(0, 10)
   921→        .map((ch) => ({
   922→          name: ch.name,
   923→          url: ch.url,
   924→          logo: ch.logo,
   925→          group: ch.group,
   926→          relevance: 0,
   927→          health: getChannelHealth(ch.url),
   928→        }));
   929→
   930→      if (generalSports.length > 0) {
   931→        return NextResponse.json({
   932→          channels: generalSports,
   933→          message: 'Chaînes sportives générales',
   934→        });
   935→      }
   936→
   937→      // Last resort: first 5 channels
   938→      return NextResponse.json({
   939→        channels: allChannels.slice(0, 5).map((ch) => ({
   940→          name: ch.name,
   941→          url: ch.url,
   942→          logo: ch.logo,
   943→          group: ch.group,
   944→          relevance: 0,
   945→          health: getChannelHealth(ch.url),
   946→        })),
   947→        message: 'Aucune chaîne spécifique trouvée',
   948→      });
   949→    }
   950→
   951→    // Build response message
   952→    const broadcasterNames = searchBroadcasters.length > 0
   953→      ? searchBroadcasters
   954→      : combinedBroadcasters.slice(0, 3).map(b => b.broadcaster);
   955→
   956→    const onlineCount = finalChannels.filter(ch => ch.health === 'online').length;
   957→    const broadcasterInfo = broadcasterNames.length > 0
   958→      ? `Diffusé sur: ${broadcasterNames.join(', ')}`
   959→      : '';
   960→
   961→    return NextResponse.json({
   962→      channels: finalChannels,
   963→      message: `${finalChannels.length} chaîne(s) trouvée(s)${onlineCount > 0 ? ` · ${onlineCount} en ligne` : ''}${broadcasterInfo ? ` · ${broadcasterInfo}` : ''}`,
   964→      broadcasters: broadcasterNames.length > 0 ? broadcasterNames : undefined,
   965→    });
   966→  } catch (error) {
   967→    console.error('[Match Stream API] Error:', error);
   968→    // Return 200 with error message so the client can display it properly
   969→    // (returning 500 causes the fetch to throw, showing "Erreur lors de la recherche")
   970→    return NextResponse.json({
   971→      channels: [],
   972→      message: 'Erreur temporaire — veuillez réessayer',
   973→      error: 'Failed to find channels',
   974→    });
   975→  }
   976→}

   981→function guessCountriesForBroadcaster(broadcaster: string, teamCountries: string[]): string[] {
