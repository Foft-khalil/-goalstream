    11→    11→
    12→    12→/**
    13→    13→ * Determine if a URL is an HLS stream (can be played with video element + hls.js)
    14→    14→ * vs an iframe embed URL (needs to be shown in an iframe)
    15→    15→ */
    16→    16→function isHlsUrl(url: string): boolean {
    17→    17→  return url.includes('.m3u8') || url.includes('m3u8');
    18→    18→}
    19→    19→
    20→    20→/**
    21→    21→ * Determine if a URL needs to go through our server-side proxy.
    22→    22→ * ALL non-HLS URLs (anything that isn't an m3u8 stream) need proxying
    23→    23→ * because they are web pages that will be blocked by X-Frame-Options
    24→    24→ * or Content-Security-Policy frame-ancestors when loaded directly in an iframe.
    25→    25→ *
    26→    26→ * The proxy fetches the page server-side, strips blocking headers,
    27→    27→ * rewrites nested iframes to also go through the proxy, and serves
    28→    28→ * the content from our own domain.
    29→    29→ */
    30→    30→function needsProxy(url: string): boolean {
    31→    31→  // HLS streams don't need proxying — they play natively in the video element
    32→    32→  if (isHlsUrl(url)) return false;
    33→    33→  // Our own proxy URLs don't need re-proxying
    34→    34→  if (url.includes('/api/proxy-stream')) return false;
    35→    35→  // Known embed sites that DON'T set X-Frame-Options → load directly in iframe.
    36→    36→  // NOTE: DaddyLive (dlive.sx) is NO LONGER in this list — it's too heavy (643KB
    37→    37→  // + ads + nested iframes) and times out. Instead, dlive.sx URLs go through
    38→    38→  // /api/proxy-stream which fetches the page server-side (stripping ads/scripts),
    39→    39→  // rewrites the nested hamis.romponalis.st iframe to also go through the proxy
    40→    40→  // (with Referer: https://dlive.sx/ so hamis returns 200), and serves a
    41→    41→  // lightweight page from our own domain.
    42→    42→  const directLoadDomains: string[] = [];
    43→    43→  if (directLoadDomains.some(d => url.includes(d))) return false;
    44→    44→  // Any other URL (web page / embed) needs proxying to bypass iframe restrictions
    45→    45→  return true;
    46→    46→}
    47→    47→
    48→    48→/**
    49→    49→ * Convert a stream URL to our proxy URL.
    50→    50→ * For HLS (.m3u8) streams, returns the URL as-is (plays in video element).
    51→    51→ * For web page / embed URLs, routes through our server-side proxy
    52→    52→ * to bypass X-Frame-Options and frame-ancestors restrictions.
    53→    53→ */
    54→    54→function getProxiedUrl(url: string): string {
    55→    55→  if (!needsProxy(url)) return url;
    56→    56→  try {
    57→    57→    const encoded = btoa(url);
    58→    58→    return `/api/proxy-stream?url=${encodeURIComponent(encoded)}`;
    59→    59→  } catch(_e) {
    60→    60→    return url;
    61→    61→  }
    62→    62→}

   348→   348→          // and retrying just wastes time. Fall back to next channel instantly.

   352→   352→              if (watchdogRef.current) { clearTimeout(watchdogRef.current); watchdogRef.current = null; }
   353→   353→              if (tryNextChannel()) return;
   354→   354→              setErrorInfo({ url: currentUrl, msg: t(language, 'player.streamUnavailable') });

   362→   362→                hls.recoverMediaError();
   363→   363→              } else {
   364→   364→                if (tryNextChannel()) return;
   365→   365→                setErrorInfo({ url: currentUrl, msg: t(language, 'player.playbackError') });
   366→   366→                hlsDestroyed = true;
   367→   367→                hls.destroy();
   368→   368→              }
   369→   369→              break;
   370→   370→            default:
   371→   371→              if (tryNextChannel()) return;
   372→   372→              setErrorInfo({ url: currentUrl, msg: t(language, 'player.playbackError') });
   373→   373→              hlsDestroyed = true;
   374→   374→              hls.destroy();
   375→   375→              break;
   376→   376→          }
   377→   377→        }
   378→   378→      });
   379→   379→
   380→   380→      hlsRef.current = hls;
   381→   381→    } else if (video.canPlayType('application/vnd.apple.mpegurl')) {
   382→   382→      video.src = currentUrl;
   383→   383→      const onLoaded = () => {
   384→   384→        setReadyUrl(currentUrl);
   385→   385→        video.play().then(() => setIsPlaying(true)).catch(() => {});
   386→   386→      };
   387→   387→      const onError = () => {
   388→   388→        if (tryNextChannel()) return;
   389→   389→        setErrorInfo({ url: currentUrl, msg: t(language, 'player.streamUnavailableShort') });
   390→   390→      };
   391→   391→      video.addEventListener('loadedmetadata', onLoaded);
   392→   392→      video.addEventListener('error', onError);
   393→   393→    } else {
   394→   394→      if (!tryNextChannel()) {
   395→   395→        queueMicrotask(() => {
   396→   396→          setErrorInfo({ url: currentUrl, msg: t(language, 'player.hlsNotSupported') });
   397→   397→        });
   398→   398→      }
   399→   399→    }
   400→   400→
   401→   401→    return () => {
   402→   402→      if (watchdogRef.current) { clearTimeout(watchdogRef.current); watchdogRef.current = null; }
   403→   403→      if (hlsRef.current) {
   404→   404→        hlsRef.current.destroy();
   405→   405→        hlsRef.current = null;
   406→   406→      }
   407→   407→    };
   408→   408→  }, [playerVisible, effectiveUrl, isHls, tryNextChannel, language]);
   409→   409→

   414→   414→      currentAltIndexRef.current = 0;
   415→   415→    } else {
   416→   416→      document.body.style.overflow = '';
   417→   417→    }
   418→   418→    return () => {
   419→   419→      document.body.style.overflow = '';
   420→   420→    };
   421→   421→  }, [playerVisible]);
   422→   422→

   427→   427→      setIsPlaying(false);
   428→   428→    } else {
   429→   429→      videoRef.current.play().then(() => setIsPlaying(true)).catch(() => {});
   430→   430→    }
   431→   431→  };
   432→   432→
   433→   433→  const toggleMute = () => {

   438→   438→
   439→   439→  const toggleFullscreen = async () => {
