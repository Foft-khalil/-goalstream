   141→   141→async function fetchAllMatches(categoryIds?: number[]): Promise<KoraMatch[]> {
   142→   142→  const cacheKey = categoryIds ? `matches-cats-${categoryIds.join(',')}` : 'matches-all';
   143→   143→
   144→   144→  const cached = getCachedMatches(cacheKey);
   145→   145→  if (cached) {
   146→   146→    console.log(`[Streams API] Cache hit for key: ${cacheKey} (${cached.length} matches)`);
   147→   147→    return cached;
   148→   148→  }
   149→   149→
   150→   150→  const allMatches: KoraMatch[] = [];
   151→   151→  let page = 1;
   152→   152→  const perPage = 200;
   153→   153→  let totalMatches = Infinity;
   154→   154→
   155→   155→  try {
   156→   156→    while ((page - 1) * perPage < totalMatches) {
   157→   157→      const url = `https://ws.kora-api.space/api/matches?per_page=${perPage}&page=${page}`;
   158→   158→      console.log(`[Streams API] Fetching page ${page} from kora-api...`);
   159→   159→
   160→   160→      const res = await fetch(url, {
   161→   161→        headers: { 'Accept': 'application/json' },
   162→   162→        signal: AbortSignal.timeout(10000),
   163→   163→      });
   164→   164→
   165→   165→      if (!res.ok) {
   166→   166→        console.warn(`[Streams API] kora-api returned HTTP ${res.status}`);
   167→   167→        break;
   168→   168→      }
   169→   169→
   170→   170→      const data: KoraMatchesResponse = await res.json();
   171→   171→      totalMatches = data.total;
   172→   172→
   173→   173→      const filtered = categoryIds
   174→   174→        ? data.data.filter((m) => categoryIds.includes(m.category.id))
   175→   175→        : data.data;
   176→   176→
   177→   177→      allMatches.push(...filtered);
   178→   178→
   179→   179→      if (data.data.length < perPage) break;
   180→   180→      page++;
   181→   181→    }
   182→   182→  } catch (err) {
   183→   183→    console.warn('[Streams API] Error fetching from kora-api:', err);
   184→   184→  }
   185→   185→
   186→   186→  console.log(`[Streams API] Fetched ${allMatches.length} matches from kora-api`);
   187→   187→  setCachedMatches(cacheKey, allMatches);
   188→   188→  return allMatches;
   189→   189→}

   450→   450→      for (const v of values) {

   456→   456→      variants.add(key);

   468→   468→function teamMatchesInName(teamVariants: string[], matchName: string): boolean {
   469→   469→  const matchLower = matchName.toLowerCase();
   470→   470→
   471→   471→  for (const variant of teamVariants) {
   472→   472→    if (!variant) continue;
   473→   473→    if (matchLower.includes(variant)) return true;
   474→   474→
   475→   475→    const variantWords = variant.split(/\s+/).filter(w => w.length > 2);
   476→   476→    if (variantWords.length > 1) {
   477→   477→      const matchCount = variantWords.filter(w => matchLower.includes(w)).length;
   478→   478→      if (matchCount >= Math.ceil(variantWords.length * 0.6)) return true;
   479→   479→    }
   480→   480→
   481→   481→    if (variant.length >= 4 && matchLower.includes(variant)) return true;
   482→   482→  }
   483→   483→
   484→   484→  return false;
   485→   485→}

   514→   514→function competitionMatchesKoraCategory(competition: string, koraCategoryName: string): boolean {
   515→   515→  const compLower = competition.toLowerCase();
   516→   516→  const catLower = koraCategoryName.toLowerCase();
   517→   517→
   518→   518→  for (const [compKey, catKeys] of Object.entries(COMPETITION_TO_KORA_CATEGORY)) {
   519→   519→    if (compLower.includes(compKey)) {
   520→   520→      return catKeys.some(ck => catLower.includes(ck) || ck.includes(catLower));
   521→   521→    }
   522→   522→  }
   523→   523→
   524→   524→  // Direct substring match between competition and category name
   525→   525→  if (compLower.length > 3 && catLower.length > 3) {
   526→   526→    if (compLower.includes(catLower) || catLower.includes(compLower)) return true;
   527→   527→  }
   528→   528→
   529→   529→  return false;
   530→   530→}

   533→   533→export async function POST(request: NextRequest) {
   534→   534→  try {
   535→   535→    const body = await request.json();
   536→   536→    const { homeTeam, awayTeam, competition, sport } = body as {
   537→   537→      homeTeam?: string;
   538→   538→      awayTeam?: string;
   539→   539→      competition?: string;
   540→   540→      sport?: string;
   541→   541→    };
   542→   542→
   543→   543→    if (!homeTeam || !awayTeam) {
   544→   544→      return NextResponse.json(
   545→   545→        { error: 'homeTeam and awayTeam are required', streams: [] },
   546→   546→        { status: 400 }
   547→   547→      );
   548→   548→    }
   549→   549→
   550→   550→    const sportLower = (sport || 'football').toLowerCase();
   551→   551→    const categoryIds = SPORT_CATEGORY_IDS[sportLower];
   552→   552→
   553→   553→    // Fetch matches from kora-api
   554→   554→    const matches = await fetchAllMatches(categoryIds);
   555→   555→
   556→   556→    // ─── Kora-API matching logic ──────────────────────────────────────────
   557→   557→    let koraStreams: Array<{
   558→   558→      name: string;
   559→   559→      url: string;
   560→   560→      lang: string;
   561→   561→      langFlag: string;
   562→   562→      source: string;
   563→   563→    }> = [];
   564→   564→    let koraMatchId: number | null = null;
   565→   565→    let koraIsLive = false;
   566→   566→    let koraMatchName: string | null = null;
   567→   567→    let koraIsExactMatch = false;
   568→   568→    let koraIsCompetitionMatch = false;
   569→   569→
   570→   570→    if (matches.length === 0) {
   571→   571→      console.log('[Streams API] No matches found from kora-api');
   572→   572→    } else {
   573→   573→      // Generate team name variants for fuzzy matching
   574→   574→      const homeVariants = getTeamVariants(homeTeam);
   575→   575→      const awayVariants = getTeamVariants(awayTeam);
   576→   576→
   577→   577→      console.log(`[Streams API] Searching for: "${homeTeam}" vs "${awayTeam}" (${sportLower})`);
   578→   578→
   579→   579→      // ─── Step 1: Try exact team name matching (best quality) ──────────────
   580→   580→      let bestMatch: KoraMatch | null = null;
   581→   581→      let bestScore = 0;
   582→   582→
   583→   583→      for (const match of matches) {
   584→   584→        let score = 0;
   585→   585→
   586→   586→        const homeInName = teamMatchesInName(homeVariants, match.name);
   587→   587→        const awayInName = teamMatchesInName(awayVariants, match.name);
   588→   588→
   589→   589→        if (homeInName) score += 50;
   590→   590→        if (awayInName) score += 50;
   591→   591→        if (match.is_live) score += 10;
   592→   592→        if (match.streams && match.streams.length > 0) score += 5;
   593→   593→        if (match.streams) score += Math.min(match.streams.length, 5);
   594→   594→
   595→   595→        if (score > bestScore && homeInName && awayInName) {
   596→   596→          bestScore = score;
   597→   597→          bestMatch = match;
   598→   598→        }
   599→   599→      }
   600→   600→
   601→   601→      // ─── Step 2: Try partial team name matching (one team matches) ────────
   602→   602→      if (!bestMatch) {
   603→   603→        for (const match of matches) {
   604→   604→          const homeInName = teamMatchesInName(homeVariants, match.name);
   605→   605→          const awayInName = teamMatchesInName(awayVariants, match.name);
   606→   606→
   607→   607→          if ((homeInName || awayInName) && match.streams && match.streams.length > 0) {
   608→   608→            const score = (homeInName ? 50 : 0) + (awayInName ? 50 : 0) + (match.is_live ? 10 : 0);
   609→   609→            if (score > bestScore) {
   610→   610→              bestScore = score;
   611→   611→              bestMatch = match;
   612→   612→            }
   613→   613→          }
   614→   614→        }
   615→   615→      }
   616→   616→
   617→   617→      // ─── Step 3: Try competition/category matching ───────────────────────
   618→   618→      if (!bestMatch && competition) {
   619→   619→        console.log(`[Streams API] Trying competition matching for: "${competition}"`);
   620→   620→
   621→   621→        let bestCompMatch: KoraMatch | null = null;
   622→   622→        let bestCompScore = 0;
   623→   623→
   624→   624→        for (const match of matches) {
   625→   625→          if (!match.streams || match.streams.length === 0) continue;
   626→   626→
   627→   627→          const catMatches = competitionMatchesKoraCategory(competition, match.category.name);
   628→   628→
   629→   629→          if (catMatches) {
   630→   630→            let score = match.streams.length * 5;
   631→   631→            if (match.is_live) score += 20; // Strongly prefer live matches
   632→   632→            score += 10; // Base score for category match
   633→   633→
   634→   634→            if (score > bestCompScore) {
   635→   635→              bestCompScore = score;
   636→   636→              bestCompMatch = match;
   637→   637→            }
   638→   638→          }
   639→   639→        }
   640→   640→
   641→   641→        if (bestCompMatch) {
   642→   642→          bestMatch = bestCompMatch;
   643→   643→          console.log(`[Streams API] Found competition match: "${bestCompMatch.name}" (${bestCompMatch.category.name})`);
   644→   644→        }
   645→   645→      }
   646→   646→
   647→   647→      // ─── Step 4: Try any live match with streams from the same sport ──────
   648→   648→      if (!bestMatch) {
   649→   649→        const liveWithStreams = matches.filter(m => m.is_live && m.streams && m.streams.length > 0);
   650→   650→        if (liveWithStreams.length > 0) {
   651→   651→          // Prefer matches from the same category if competition matches
   652→   652→          if (competition) {
   653→   653→            const categoryMatch = liveWithStreams.find(m =>
   654→   654→              competitionMatchesKoraCategory(competition, m.category.name)
   655→   655→            );
   656→   656→            if (categoryMatch) bestMatch = categoryMatch;
   657→   657→          }
   658→   658→          // If still no match, take the first live match with most streams
   659→   659→          if (!bestMatch) {
   660→   660→            bestMatch = liveWithStreams.sort((a, b) => (b.streams?.length || 0) - (a.streams?.length || 0))[0];
   661→   661→          }
   662→   662→        }
   663→   663→      }
   664→   664→
   665→   665→      if (bestMatch && bestMatch.streams && bestMatch.streams.length > 0) {
   666→   666→        // Build the streams response with language flags
   667→   667→        koraStreams = bestMatch.streams.map((stream) => ({
   668→   668→          name: `${stream.lang || 'Unknown'} Stream`,
   669→   669→          url: stream.url,
   670→   670→          lang: stream.lang || 'Unknown',
   671→   671→          langFlag: LANG_FLAGS[stream.lang] || '🌐',
   672→   672→          source: 'kora-api',
   673→   673→        }));
   674→   674→
   675→   675→        koraMatchId = bestMatch.id;
   676→   676→        koraIsLive = bestMatch.is_live;
   677→   677→        koraMatchName = bestMatch.name;
   678→   678→
   679→   679→        // Determine match quality for the response
   680→   680→        koraIsExactMatch = bestScore >= 100; // Both teams matched
   681→   681→        koraIsCompetitionMatch = bestScore < 100 && bestScore > 0;
   682→   682→
   683→   683→        console.log(`[Streams API] kora-api match: "${bestMatch.name}" with ${koraStreams.length} streams (live: ${bestMatch.is_live}, exact: ${koraIsExactMatch})`);
   684→   684→      } else {
   685→   685→        console.log(`[Streams API] No matching match found from kora-api for "${homeTeam}" vs "${awayTeam}"`);
   686→   686→      }
   687→   687→    }
   688→   688→
   689→   689→    // ─── Fetch rojadirecta streams ────────────────────────────────────────
   690→   690→    const rojaStreams = await fetchRojaDirectaStreams(homeTeam, awayTeam);
   691→   691→
   692→   692→    // ─── Merge results ────────────────────────────────────────────────────
   693→   693→    const allStreams = [...koraStreams, ...rojaStreams];
   694→   694→
   695→   695→    // ─── Sort streams: prioritize resolvable domains (fltvhd.com → m3u8) ──
   696→   696→    // Known domains that resolve to m3u8 via our resolve-stream API
   697→   697→    const RESOLVABLE_DOMAINS = ['fltvhd.com', 'streams.center', 'streamcenter.pro'];
   698→   698→    // Known domains that use obfuscated JS and can't be resolved server-side
   699→   699→    const UNRESOLVABLE_DOMAINS = ['tvhd1.com'];
   700→   700→
   701→   701→    allStreams.sort((a, b) => {
   702→   702→      const aResolvable = RESOLVABLE_DOMAINS.some(d => a.url.includes(d));
   703→   703→      const bResolvable = RESOLVABLE_DOMAINS.some(d => b.url.includes(d));
   704→   704→      const aUnresolvable = UNRESOLVABLE_DOMAINS.some(d => a.url.includes(d));
   705→   705→      const bUnresolvable = UNRESOLVABLE_DOMAINS.some(d => b.url.includes(d));
   706→   706→      const aIsM3u8 = a.url.includes('.m3u8') || a.url.includes('m3u8');
   707→   707→      const bIsM3u8 = b.url.includes('.m3u8') || b.url.includes('m3u8');
   708→   708→
   709→   709→      // Direct m3u8 URLs are best
   710→   710→      if (aIsM3u8 && !bIsM3u8) return -1;
   711→   711→      if (!aIsM3u8 && bIsM3u8) return 1;
   712→   712→      // Resolvable domains next
   713→   713→      if (aResolvable && !bResolvable) return -1;
   714→   714→      if (!aResolvable && bResolvable) return 1;
   715→   715→      // Unresolvable domains last
   716→   716→      if (!aUnresolvable && bUnresolvable) return -1;
   717→   717→      if (aUnresolvable && !bUnresolvable) return 1;
   718→   718→      return 0;
   719→   719→    });
   720→   720→
   721→   721→    // Determine primary source
   722→   722→    let primarySource = 'kora-api';
   723→   723→    if (koraStreams.length === 0 && rojaStreams.length > 0) {
   724→   724→      primarySource = 'rojadirecta';
   725→   725→    } else if (koraStreams.length > 0 && rojaStreams.length > 0) {
   726→   726→      primarySource = 'kora-api+rojadirecta';
   727→   727→    }
   728→   728→
   729→   729→    if (allStreams.length === 0) {
   730→   730→      console.log(`[Streams API] No streams found from any source for "${homeTeam}" vs "${awayTeam}"`);
   731→   731→      return NextResponse.json({
   732→   732→        streams: [],
   733→   733→        matchId: koraMatchId,
   734→   734→        matchName: koraMatchName,
   735→   735→        isLive: koraIsLive,
   736→   736→        source: 'kora-api',
   737→   737→      });
   738→   738→    }
   739→   739→
   740→   740→    console.log(`[Streams API] Total: ${allStreams.length} streams (kora-api: ${koraStreams.length}, rojadirecta: ${rojaStreams.length}) for "${homeTeam}" vs "${awayTeam}"`);
   741→   741→
   742→   742→    return NextResponse.json({
   743→   743→      streams: allStreams,
   744→   744→      matchId: koraMatchId,
   745→   745→      matchName: koraMatchName,
   746→   746→      isLive: koraIsLive,
   747→   747→      isExactMatch: koraIsExactMatch,
   748→   748→      isCompetitionMatch: koraIsCompetitionMatch,
   749→   749→      source: primarySource,
   750→   750→    });
   751→   751→  } catch (err) {
   752→   752→    console.error('[Streams API] Error:', err);
   753→   753→    return NextResponse.json(
   754→   754→      {
   755→   755→        streams: [],
   756→   756→        matchId: null,
   757→   757→        isLive: false,
   758→   758→        source: 'kora-api',

   760→   760→      },
   761→   761→      { status: 200 }
   762→   762→    );
   763→   763→  }
   764→   764→}
