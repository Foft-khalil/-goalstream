    10→    10→type StreamStatus = 'loading' | 'ready' | 'error';
    11→    11→
    12→    12→/**
    13→    13→ * Determine if a URL is an HLS stream (can be played with video element + hls.js)
    14→    14→ * vs an iframe embed URL (needs to be shown in an iframe)
    15→    15→ */
    16→    16→function isHlsUrl(url: string): boolean {
    17→    17→  return url.includes('.m3u8') || url.includes('m3u8');
    18→    18→}
    19→    19→
    20→    20→/**
    21→    21→ * Check if a URL is a kora-api/embed-style stream that needs proxying.
    22→    22→ * These are URLs from providers like streams.center that contain nested iframes
    23→    23→ * and won't work directly in a sandboxed iframe.
    24→    24→ */
    25→    25→function needsProxy(url: string): boolean {
    26→    26→  const proxyDomains = [
    27→    27→    'streams.center',
    28→    28→    'kora-api.top',
    29→    29→    '000007.mov',
    30→    30→    'streamcenter.pro',
    31→    31→    'tvtvhd.com',
    32→    32→  ];
    33→    33→  return proxyDomains.some(domain => url.includes(domain));
    34→    34→}
    35→    35→
    36→    36→/**
    37→    37→ * Convert a direct stream URL to our proxy URL.
    38→    38→ * This allows embed-style streams to work in our iframe by serving
    39→    39→ * them from our own domain (same method used by us-sport.eu).
    40→    40→ */
    41→    41→function getProxiedUrl(url: string): string {
    42→    42→  if (!needsProxy(url)) return url;
    43→    43→  try {
    44→    44→    const encoded = btoa(url);
    45→    45→    return `/api/proxy-stream?url=${encodeURIComponent(encoded)}`;
    46→    46→  } catch(_e) {
    47→    47→    return url;
    48→    48→  }
    49→    49→}

    55→    55→export default function VideoPlayer() {

    58→    58→    useAppStore();

    60→    60→  const hlsRef = useRef<Hls | null>(null);
    61→    61→  const containerRef = useRef<HTMLDivElement>(null);
    62→    62→  const [isPlaying, setIsPlaying] = useState(false);

    64→    64→  const [isFullscreen, setIsFullscreen] = useState(false);
    65→    65→  const [showControls, setShowControls] = useState(true);
    66→    66→  const retryCountRef = useRef(0);
    67→    67→  const controlsTimeoutRef = useRef<NodeJS.Timeout>();
    68→    68→  const currentAltIndexRef = useRef(0);
    69→    69→
    70→    70→  // Track ready/error state with associated URL so they auto-reset when URL changes
    71→    71→  const [readyUrl, setReadyUrl] = useState<string>('');
    72→    72→  const [errorInfo, setErrorInfo] = useState<{ url: string; msg: string } | null>(null);
    73→    73→
    74→    74→  // Resolve-stream state: tracks whether we resolved an iframe URL to m3u8
    75→    75→  const [resolvedUrl, setResolvedUrl] = useState<string | null>(null);
    76→    76→  const [isResolving, setIsResolving] = useState(false);
    77→    77→
    78→    78→  // Iframe error/timeout state
    79→    79→  const [iframeError, setIframeError] = useState(false);
    80→    80→  const [iframeTimedOut, setIframeTimedOut] = useState(false);
    81→    81→  const iframeTimerRef = useRef<NodeJS.Timeout>();
    82→    82→
    83→    83→  // The effective URL to play (resolved m3u8 takes priority over original URL)

    89→    89→
    90→    90→  // For iframe streams, compute the actual URL to use (proxied if needed)

    94→    94→  }, [isIframe, effectiveUrl]);
    95→    95→
    96→    96→  // For iframe streams, compute ready state directly instead of using an effect
    97→    97→  const iframeReady = isIframe && !!effectiveUrl;
    98→    98→
    99→    99→  // Compute derived state - auto-reset when URL changes
   100→   100→  const streamReady = (isIframe ? iframeReady : (readyUrl === effectiveUrl && !!effectiveUrl));
   101→   101→  const streamError = errorInfo?.url === effectiveUrl ? errorInfo.msg : null;
   102→   102→  const isLoading = isHls && playerVisible && !!effectiveUrl && !streamReady && !streamError;
   103→   103→
   104→   104→  // Reset resolve/iframe state when the original URL changes
   105→   105→  useEffect(() => {
   106→   106→    setResolvedUrl(null);
   107→   107→    setIsResolving(false);
   108→   108→    setIframeError(false);
   109→   109→    setIframeTimedOut(false);
   110→   110→  }, [playerStreamUrl]);
   111→   111→
   112→   112→  // Resolve-stream: try to resolve iframe URLs to direct m3u8 in the background
   113→   113→  // Don't block initial render — show iframe immediately while resolving
   114→   114→  useEffect(() => {

   116→   116→    // Only try resolving if it's an iframe URL (not already HLS)

   118→   118→    // Don't re-resolve if we already have a resolved URL for this stream
   119→   119→    if (resolvedUrl) return;

   122→   122→    setIsResolving(true);
   123→   123→

   125→   125→      try {
   126→   126→        const res = await fetch('/api/resolve-stream', {
   127→   127→          method: 'POST',
   128→   128→          headers: { 'Content-Type': 'application/json' },
   129→   129→          body: JSON.stringify({ url: playerStreamUrl }),
   130→   130→        });

   132→   132→        if (!res.ok) {

   135→   135→        }
   136→   136→
   137→   137→        const data = await res.json();
   138→   138→        if (cancelled) return;
   139→   139→
   140→   140→        if (data.resolved && data.type === 'hls' && data.resolvedUrl) {
   141→   141→          console.log('[VideoPlayer] Resolved to m3u8:', data.resolvedUrl);
   142→   142→          setResolvedUrl(data.resolvedUrl);
   143→   143→        } else {
   144→   144→          console.log('[VideoPlayer] Could not resolve to m3u8, falling back to iframe');
   145→   145→        }
   146→   146→      } catch (err) {
   147→   147→        if (cancelled) return;
   148→   148→        console.warn('[VideoPlayer] resolve-stream error:', err);
   149→   149→      } finally {
   150→   150→        if (!cancelled) setIsResolving(false);
   151→   151→      }
   152→   152→    };
   153→   153→
   154→   154→    resolveStream();
   155→   155→    return () => { cancelled = true; };
   156→   156→  }, [playerVisible, playerStreamUrl]);
   157→   157→
   158→   158→  // Iframe timeout: show warning if iframe hasn't loaded after 15 seconds
   159→   159→  useEffect(() => {
   160→   160→    if (!isIframe || !playerVisible || !effectiveUrl) return;
   161→   161→
   162→   162→    setIframeError(false);
   163→   163→    setIframeTimedOut(false);
   164→   164→
   165→   165→    iframeTimerRef.current = setTimeout(() => {
   166→   166→      setIframeTimedOut(true);
   167→   167→    }, 15000);
   168→   168→
   169→   169→    return () => {
   170→   170→      if (iframeTimerRef.current) {
   171→   171→        clearTimeout(iframeTimerRef.current);
   172→   172→      }
   173→   173→    };
   174→   174→  }, [isIframe, playerVisible, effectiveUrl]);
   175→   175→
   176→   176→  const hideControlsAfterDelay = useCallback(() => {
   177→   177→    if (controlsTimeoutRef.current) clearTimeout(controlsTimeoutRef.current);
   178→   178→    setShowControls(true);
   179→   179→    if (isPlaying) {
   180→   180→      controlsTimeoutRef.current = setTimeout(() => setShowControls(false), 3000);
   181→   181→    }
   182→   182→  }, [isPlaying]);
   183→   183→
   184→   184→  // Auto-switch to next alternative channel
   185→   185→  const tryNextChannel = useCallback(() => {
   186→   186→    const alternatives = playerAlternatives;
   187→   187→    if (!alternatives || alternatives.length === 0) return false;
   188→   188→
   189→   189→    if (currentAltIndexRef.current < alternatives.length) {
   190→   190→      const next = alternatives[currentAltIndexRef.current];
   191→   191→      currentAltIndexRef.current += 1;
   192→   192→      console.log(`[VideoPlayer] Auto-switching to: ${next.name}`);
   193→   193→      retryCountRef.current = 0;
   194→   194→      openPlayer(next.url, next.name, next.logo || undefined, alternatives.filter((a) => a.url !== next.url));
   195→   195→      return true;
   196→   196→    }
   197→   197→    return false;
   198→   198→  }, [playerAlternatives, openPlayer]);
   199→   199→
   200→   200→  // Setup HLS player (only for HLS URLs, including resolved m3u8)
   201→   201→  useEffect(() => {
   202→   202→    if (!playerVisible || !videoRef.current || !effectiveUrl || !isHls) return;
   203→   203→
   204→   204→    const video = videoRef.current;
   205→   205→    const currentUrl = effectiveUrl;
   206→   206→
   207→   207→    // State auto-resets when URL changes (via derived computation above)
   208→   208→    retryCountRef.current = 0;
   209→   209→
   210→   210→    // Destroy previous HLS instance
   211→   211→    if (hlsRef.current) {
   212→   212→      hlsRef.current.destroy();
   213→   213→      hlsRef.current = null;
   214→   214→    }
   215→   215→
   216→   216→    if (Hls.isSupported()) {
   217→   217→      const hls = new Hls({
   218→   218→        enableWorker: true,
   219→   219→        lowLatencyMode: true,
   220→   220→        maxBufferLength: 30,
   221→   221→        maxMaxBufferLength: 60,
   222→   222→        startLevel: -1,
   223→   223→        manifestLoadingTimeOut: 10000,
   224→   224→        manifestLoadingMaxRetry: 1,
   225→   225→        levelLoadingTimeOut: 10000,
   226→   226→        levelLoadingMaxRetry: 1,

   228→   228→        fragLoadingMaxRetry: 1,

   230→   230→
   231→   231→      hls.loadSource(currentUrl);
   232→   232→      hls.attachMedia(video);
   233→   233→
   234→   234→      hls.on(Hls.Events.MANIFEST_PARSED, () => {
   235→   235→        setReadyUrl(currentUrl);
   236→   236→        video.play().then(() => setIsPlaying(true)).catch(() => {});

   238→   238→
   239→   239→      hls.on(Hls.Events.ERROR, (_event, data) => {

   242→   242→            case Hls.ErrorTypes.NETWORK_ERROR:
   243→   243→              if (retryCountRef.current < 1) {
   244→   244→                retryCountRef.current += 1;
   245→   245→                hls.startLoad();
   246→   246→              } else {
   247→   247→                if (tryNextChannel()) return;
   248→   248→                setErrorInfo({ url: currentUrl, msg: t(language, 'player.streamUnavailable') });
   249→   249→                hls.destroy();

   251→   251→              break;
   252→   252→            case Hls.ErrorTypes.MEDIA_ERROR:
   253→   253→              hls.recoverMediaError();
   254→   254→              break;
   255→   255→            default:
   256→   256→              if (tryNextChannel()) return;
   257→   257→              setErrorInfo({ url: currentUrl, msg: t(language, 'player.playbackError') });
   258→   258→              hls.destroy();
   259→   259→              break;
   260→   260→          }
   261→   261→        }
   262→   262→      });
   263→   263→
   264→   264→      hlsRef.current = hls;
   265→   265→    } else if (video.canPlayType('application/vnd.apple.mpegurl')) {
   266→   266→      video.src = currentUrl;
   267→   267→      const onLoaded = () => {
   268→   268→        setReadyUrl(currentUrl);
   269→   269→        video.play().then(() => setIsPlaying(true)).catch(() => {});
   270→   270→      };
   271→   271→      const onError = () => {
   272→   272→        if (tryNextChannel()) return;
   273→   273→        setErrorInfo({ url: currentUrl, msg: t(language, 'player.streamUnavailableShort') });
   274→   274→      };
   275→   275→      video.addEventListener('loadedmetadata', onLoaded);
   276→   276→      video.addEventListener('error', onError);
   277→   277→    } else {
   278→   278→      if (!tryNextChannel()) {
   279→   279→        queueMicrotask(() => {
   280→   280→          setErrorInfo({ url: currentUrl, msg: t(language, 'player.hlsNotSupported') });
   281→   281→        });
   282→   282→      }
   283→   283→    }
   284→   284→
   285→   285→    return () => {
   286→   286→      if (hlsRef.current) {
   287→   287→        hlsRef.current.destroy();
   288→   288→        hlsRef.current = null;
   289→   289→      }
   290→   290→    };
   291→   291→  }, [playerVisible, effectiveUrl, isHls, tryNextChannel, language]);
   292→   292→
   293→   293→  // Lock body scroll when player is open
   294→   294→  useEffect(() => {
   295→   295→    if (playerVisible) {
   296→   296→      document.body.style.overflow = 'hidden';
   297→   297→      currentAltIndexRef.current = 0;
   298→   298→    } else {
   299→   299→      document.body.style.overflow = '';
   300→   300→    }
   301→   301→    return () => {
   302→   302→      document.body.style.overflow = '';
   303→   303→    };
   304→   304→  }, [playerVisible]);
   305→   305→
   306→   306→  const togglePlay = () => {
   307→   307→    if (!videoRef.current) return;
   308→   308→    if (isPlaying) {
   309→   309→      videoRef.current.pause();
   310→   310→      setIsPlaying(false);
   311→   311→    } else {
   312→   312→      videoRef.current.play().then(() => setIsPlaying(true)).catch(() => {});
   313→   313→    }
   314→   314→  };
   315→   315→
   316→   316→  const toggleMute = () => {
   317→   317→    if (!videoRef.current) return;
   318→   318→    videoRef.current.muted = !isMuted;
   319→   319→    setIsMuted(!isMuted);
   320→   320→  };
   321→   321→
   322→   322→  const toggleFullscreen = async () => {
   323→   323→    if (!containerRef.current) return;
   324→   324→    if (!isFullscreen) {
   325→   325→      await containerRef.current.requestFullscreen();
   326→   326→      setIsFullscreen(true);
   327→   327→    } else {
   328→   328→      await document.exitFullscreen();
   329→   329→      setIsFullscreen(false);
   330→   330→    }
   331→   331→  };
   332→   332→
   333→   333→  const handleRetry = () => {
   334→   334→    retryCountRef.current = 0;
   335→   335→    setReadyUrl('');
   336→   336→    setErrorInfo(null);
   337→   337→    setResolvedUrl(null);
   338→   338→    setIframeError(false);
   339→   339→    setIframeTimedOut(false);
   340→   340→  };
   341→   341→
   342→   342→  const handleSwitchChannel = (channel: { url: string; name: string; logo?: string }) => {
   343→   343→    retryCountRef.current = 0;
   344→   344→    openPlayer(channel.url, channel.name, channel.logo || undefined);
   345→   345→  };
   346→   346→
   347→   347→  /** Manually try to resolve the current iframe URL to a direct m3u8 stream */
   348→   348→  const handleTryResolve = async () => {
   349→   349→    if (!playerStreamUrl || isResolving) return;
   350→   350→    setIframeError(false);
   351→   351→    setIframeTimedOut(false);
   352→   352→    setIsResolving(true);
   353→   353→    try {
   354→   354→      const res = await fetch('/api/resolve-stream', {
   355→   355→        method: 'POST',
   356→   356→        headers: { 'Content-Type': 'application/json' },
   357→   357→        body: JSON.stringify({ url: playerStreamUrl }),
   358→   358→      });
   359→   359→      const data = await res.json();
   360→   360→      if (data.resolved && data.type === 'hls' && data.resolvedUrl) {
   361→   361→        setResolvedUrl(data.resolvedUrl);
   362→   362→      }
   363→   363→    } catch(_e) {
   364→   364→      // Silently fail — user can still use iframe or external sites
   365→   365→    } finally {
   366→   366→      setIsResolving(false);
   367→   367→    }
   368→   368→  };
   369→   369→
   370→   370→  /** Render external site fallback buttons (SportStream & RojaDirecta) */
   371→   371→  const renderExternalSiteButtons = () => (
   372→   372→    <div className="flex flex-col gap-1.5 w-full max-w-xs mt-2">
   373→   373→      <p className="text-white/50 text-xs mb-1">{t(language, 'player.watchElsewhere')}</p>
   374→   374→      <a
   375→   375→        href={SPORTSTREAM_URL}
   376→   376→        target="_blank"
   377→   377→        rel="noopener noreferrer"
   378→   378→        className="w-full flex items-center gap-2 px-3 py-2.5 bg-white/5 hover:bg-white/15 rounded-lg transition-colors text-left"
   379→   379→      >
   380→   380→        <Globe className="h-4 w-4 text-green-400 shrink-0" />
   381→   381→        <span className="text-white text-sm truncate flex-1">SportStream</span>
   382→   382→        <ExternalLink className="h-3.5 w-3.5 text-white/40 shrink-0" />
   383→   383→      </a>
   384→   384→      <a
   385→   385→        href={ROJADIRECTA_URL}
   386→   386→        target="_blank"
   387→   387→        rel="noopener noreferrer"
   388→   388→        className="w-full flex items-center gap-2 px-3 py-2.5 bg-white/5 hover:bg-white/15 rounded-lg transition-colors text-left"
   389→   389→      >
   390→   390→        <Globe className="h-4 w-4 text-red-400 shrink-0" />
   391→   391→        <span className="text-white text-sm truncate flex-1">RojaDirecta</span>
   392→   392→        <ExternalLink className="h-3.5 w-3.5 text-white/40 shrink-0" />
   393→   393→      </a>
   394→   394→    </div>
   395→   395→  );
   396→   396→
   397→   397→  if (!playerVisible) return null;
   398→   398→
   399→   399→  return (
   400→   400→    <div className="fixed inset-0 z-50 bg-black flex flex-col">
   401→   401→      {/* Header */}
   402→   402→      <div
   403→   403→        className={`absolute top-0 left-0 right-0 z-10 bg-gradient-to-b from-black/80 to-transparent p-4 transition-opacity duration-300 ${showControls || isIframe ? 'opacity-100' : 'opacity-0'}`}
   404→   404→        onMouseMove={hideControlsAfterDelay}
   405→   405→      >
   406→   406→        <div className="flex items-center gap-3">
   407→   407→          <Button
   408→   408→            variant="ghost"
   409→   409→            size="icon"
   410→   410→            onClick={closePlayer}
   411→   411→            className="text-white hover:bg-white/20"
   412→   412→          >
   413→   413→            <X className="h-5 w-5" />
   414→   414→          </Button>
   415→   415→          {playerChannelLogo && (
   416→   416→            <img
   417→   417→              src={playerChannelLogo}
   418→   418→              alt=""
   419→   419→              className="w-8 h-8 rounded object-contain"
   420→   420→              onError={(e) => {
   421→   421→                (e.target as HTMLImageElement).style.display = 'none';
   422→   422→              }}
   423→   423→            />
   424→   424→          )}
   425→   425→          <h2 className="text-white font-semibold text-lg truncate">{playerChannelName}</h2>
   426→   426→          {isIframe && !isResolving && (
   427→   427→            <span className="text-green-400/80 text-xs ml-2 flex items-center gap-1">
   428→   428→              <ExternalLink className="h-3 w-3" />
   429→   429→              {t(language, 'player.live')}
   430→   430→            </span>
   431→   431→          )}
   432→   432→          {isResolving && (
   433→   433→            <span className="text-amber-400/80 text-xs ml-2 flex items-center gap-1">

   435→   435→              {t(language, 'player.resolvingStream')}
   436→   436→            </span>
   437→   437→          )}
   438→   438→          {playerAlternatives && playerAlternatives.length > 0 && (

   441→   441→            </span>

   444→   444→      </div>
   445→   445→
   446→   446→      {/* Video / Iframe content */}
   447→   447→      <div

   450→   450→        onMouseMove={hideControlsAfterDelay}
   451→   451→        onClick={isHls ? togglePlay : undefined}
   452→   452→      >
   453→   453→        {isIframe ? (
   454→   454→          /* Iframe-based stream (e.g. from kora-api via proxy) */
   455→   455→          <iframe

   457→   457→            className="w-full h-full border-0"

   459→   459→            allowFullScreen
   460→   460→            title={`${t(language, 'player.liveStream')}: ${playerChannelName}`}
   461→   461→            referrerPolicy="no-referrer"
   462→   462→            onError={() => {
   463→   463→              console.warn('[VideoPlayer] iframe onError triggered');
   464→   464→              setIframeError(true);
   465→   465→            }}
   466→   466→          />
   467→   467→        ) : (
