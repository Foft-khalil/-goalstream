    13→/** Get YYYYMMDD string for a Date */
    14→function formatDateYMD(date: Date): string {
    15→  const y = date.getFullYear();
    16→  const m = String(date.getMonth() + 1).padStart(2, '0');
    17→  const d = String(date.getDate()).padStart(2, '0');
    18→  return `${y}${m}${d}`;
    19→}
    20→
    21→/** Get the YYYYMMDD for a given DateTab relative to today */
    22→function getDateForTab(tab: DateTab): string {
    23→  const offset = parseInt(tab.replace('day', ''), 10);
    24→  const now = new Date();
    25→  const target = new Date(now.getTime() + offset * 24 * 60 * 60 * 1000);
    26→  return formatDateYMD(target);
    27→}
    28→
    29→/** Check if a match's ISO date falls on a given YYYYMMDD date */
    30→function isMatchOnDate(matchDate: string | null, ymd: string): boolean {
    31→  if (!matchDate) return false;
    32→  try {
    33→    const d = new Date(matchDate);
    34→    return formatDateYMD(d) === ymd;
    35→  } catch {
    36→    return false;
    37→  }
    38→}
    39→
    40→/** Format a countdown to a future date */
    41→function formatCountdown(targetDate: Date, language: string): string {
    42→  const now = new Date();
    43→  const diffMs = targetDate.getTime() - now.getTime();
    44→  if (diffMs <= 0) return '';
    45→
    46→  const diffMins = Math.floor(diffMs / 60000);
    47→  const diffHours = Math.floor(diffMins / 60);
    48→  const diffDays = Math.floor(diffHours / 24);
    49→
    50→  if (diffDays > 0) {
    51→    const remainingHours = diffHours % 24;
    52→    if (remainingHours > 0) {
    53→      return `${diffDays}j ${remainingHours}h`;
    54→    }
    55→    return `${diffDays}j`;
    56→  }
    57→  if (diffHours > 0) {
    58→    const remainingMins = diffMins % 60;
    59→    if (remainingMins > 0) {
    60→      return `${diffHours}h ${remainingMins}min`;
    61→    }
    62→    return `${diffHours}h`;
    63→  }
    64→  return `${diffMins}min`;
    65→}
    66→
    67→/** Format match time from ISO date */
    68→function formatMatchTime(matchDate: string | null, language: string): string {
    69→  if (!matchDate) return '';
    70→  try {
    71→    const d = new Date(matchDate);
    72→    const hours = String(d.getHours()).padStart(2, '0');
    73→    const mins = String(d.getMinutes()).padStart(2, '0');
    74→    return `${hours}:${mins}`;
    75→  } catch {
    76→    return '';
    77→  }
    78→}
    79→
    80→/** Get very short weekday label (1-2 chars) for compact display */
    81→function getMiniWeekday(date: Date, lang: Language): string {
    82→  const weekdays = translations[lang].dates.weekdaysShort;
    83→  const day = weekdays[date.getDay()];
    84→  // Take first 2-3 chars for compact display
    85→  return day.replace('.', '').substring(0, 3);
    86→}

    88→/** Date range: 3 days back + today + 30 days forward = 34 days total */
    89→const DAYS_BACK = 3;
    90→const DAYS_FORWARD = 30;
    91→
    92→/** Generate all date tab keys dynamically */
    93→function generateDateTabs(): DateTab[] {
    94→  const tabs: DateTab[] = [];
    95→  for (let i = -DAYS_BACK; i <= DAYS_FORWARD; i++) {
    96→    tabs.push(`day${i}`);
    97→  }
    98→  return tabs;
    99→}
