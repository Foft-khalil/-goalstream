    10→    10→type StreamStatus = 'loading' | 'ready' | 'error';
    11→    11→
    12→    12→/**
    13→    13→ * Determine if a URL is an HLS stream (can be played with video element + hls.js)
    14→    14→ * vs an iframe embed URL (needs to be shown in an iframe)
    15→    15→ */
    16→    16→function isHlsUrl(url: string): boolean {
    17→    17→  return url.includes('.m3u8') || url.includes('m3u8');
    18→    18→}
    19→    19→
    20→    20→/**
    21→    21→ * Check if a URL is a kora-api/embed-style stream that needs proxying.
    22→    22→ * These are URLs from providers like streams.center that contain nested iframes
    23→    23→ * and won't work directly in a sandboxed iframe.
    24→    24→ */
    25→    25→function needsProxy(url: string): boolean {
    26→    26→  const proxyDomains = [
    27→    27→    'streams.center',
    28→    28→    'kora-api.top',
    29→    29→    '000007.mov',
    30→    30→    'streamcenter.pro',
    31→    31→    'tvtvhd.com',
    32→    32→    'go4score.app',
    33→    33→    'smartagro.mov',
    34→    34→    'hes-goal.eu',
    35→    35→    'goalz.zip',
    36→    36→  ];
    37→    37→  return proxyDomains.some(domain => url.includes(domain));
    38→    38→}
    39→    39→
    40→    40→/**
    41→    41→ * Convert a direct stream URL to our proxy URL.
    42→    42→ * This allows embed-style streams to work in our iframe by serving
    43→    43→ * them from our own domain (same method used by us-sport.eu).
    44→    44→ */
    45→    45→function getProxiedUrl(url: string): string {
    46→    46→  if (!needsProxy(url)) return url;
    47→    47→  try {
    48→    48→    const encoded = btoa(url);
    49→    49→    return `/api/proxy-stream?url=${encodeURIComponent(encoded)}`;
    50→    50→  } catch(_e) {
    51→    51→    return url;
    52→    52→  }
    53→    53→}

    55→    55→export default function VideoPlayer() {

    57→    57→    useAppStore();

    63→    63→  const [isFullscreen, setIsFullscreen] = useState(false);

    72→    72→
    73→    73→  // Resolve-stream state: tracks whether we resolved an iframe URL to m3u8

    76→    76→
    77→    77→  // Iframe error/timeout state

    79→    79→  const [iframeTimedOut, setIframeTimedOut] = useState(false);
    80→    80→  const iframeTimerRef = useRef<NodeJS.Timeout>();
    81→    81→

    88→    88→
    89→    89→  // For iframe streams, compute the actual URL to use (proxied if needed)

    93→    93→  }, [isIframe, effectiveUrl]);
    94→    94→
    95→    95→  // For iframe streams, compute ready state directly instead of using an effect
    96→    96→  const iframeReady = isIframe && !!effectiveUrl;
    97→    97→
    98→    98→  // Compute derived state - auto-reset when URL changes
    99→    99→  const streamReady = (isIframe ? iframeReady : (readyUrl === effectiveUrl && !!effectiveUrl));
   100→   100→  const streamError = errorInfo?.url === effectiveUrl ? errorInfo.msg : null;
   101→   101→  const isLoading = isHls && playerVisible && !!effectiveUrl && !streamReady && !streamError;
   102→   102→
   103→   103→  // Reset resolve/iframe state when the original URL changes
   104→   104→  useEffect(() => {
   105→   105→    setResolvedUrl(null);
   106→   106→    setIsResolving(false);
   107→   107→    setIframeError(false);
   108→   108→    setIframeTimedOut(false);
   109→   109→  }, [playerStreamUrl]);
   110→   110→
   111→   111→  // Resolve-stream: try to resolve iframe URLs to direct m3u8 in the background

   114→   114→    // Only try resolving if it's an iframe URL (not already HLS)
   115→   115→    if (isHlsUrl(playerStreamUrl)) return;

   117→   117→    if (resolvedUrl) return;

   120→   120→    setIsResolving(true);
   121→   121→

   124→   124→        const res = await fetch('/api/resolve-stream', {
   125→   125→          method: 'POST',
   126→   126→          headers: { 'Content-Type': 'application/json' },
   127→   127→          body: JSON.stringify({ url: playerStreamUrl }),
   128→   128→        });

   132→   132→          return;
   133→   133→        }
   134→   134→
   135→   135→        const data = await res.json();
   136→   136→        if (cancelled) return;
   137→   137→
   138→   138→        if (data.resolved && data.type === 'hls' && data.resolvedUrl) {
   139→   139→          console.log('[VideoPlayer] Resolved to m3u8:', data.resolvedUrl);
   140→   140→          setResolvedUrl(data.resolvedUrl);
   141→   141→        } else {
   142→   142→          console.log('[VideoPlayer] Could not resolve to m3u8, falling back to iframe');
   143→   143→        }
   144→   144→      } catch (err) {
   145→   145→        if (cancelled) return;
   146→   146→        console.warn('[VideoPlayer] resolve-stream error:', err);
   147→   147→      } finally {
   148→   148→        if (!cancelled) setIsResolving(false);
   149→   149→      }
   150→   150→    };
   151→   151→
   152→   152→    resolveStream();
   153→   153→    return () => { cancelled = true; };
   154→   154→  }, [playerVisible, playerStreamUrl]);
   155→   155→
   156→   156→  // Iframe timeout: show warning if iframe hasn't loaded after 15 seconds
   157→   157→  useEffect(() => {
   158→   158→    if (!isIframe || !playerVisible || !effectiveUrl) return;
   159→   159→
   160→   160→    setIframeError(false);
   161→   161→    setIframeTimedOut(false);
   162→   162→
   163→   163→    iframeTimerRef.current = setTimeout(() => {
   164→   164→      setIframeTimedOut(true);
   165→   165→    }, 15000);
   166→   166→
   167→   167→    return () => {
   168→   168→      if (iframeTimerRef.current) {
   169→   169→        clearTimeout(iframeTimerRef.current);
   170→   170→      }
   171→   171→    };
   172→   172→  }, [isIframe, playerVisible, effectiveUrl]);
   173→   173→
   174→   174→  const hideControlsAfterDelay = useCallback(() => {
   175→   175→    if (controlsTimeoutRef.current) clearTimeout(controlsTimeoutRef.current);
   176→   176→    setShowControls(true);
   177→   177→    if (isPlaying) {
   178→   178→      controlsTimeoutRef.current = setTimeout(() => setShowControls(false), 3000);
   179→   179→    }
   180→   180→  }, [isPlaying]);
   181→   181→
   182→   182→  // Auto-switch to next alternative channel
   183→   183→  const tryNextChannel = useCallback(() => {
   184→   184→    const alternatives = playerAlternatives;
   185→   185→    if (!alternatives || alternatives.length === 0) return false;
   186→   186→
   187→   187→    if (currentAltIndexRef.current < alternatives.length) {
   188→   188→      const next = alternatives[currentAltIndexRef.current];
   189→   189→      currentAltIndexRef.current += 1;
   190→   190→      console.log(`[VideoPlayer] Auto-switching to: ${next.name}`);
   191→   191→      retryCountRef.current = 0;
   192→   192→      openPlayer(next.url, next.name, next.logo || undefined, alternatives.filter((a) => a.url !== next.url));
   193→   193→      return true;
   194→   194→    }
   195→   195→    return false;
   196→   196→  }, [playerAlternatives, openPlayer]);
   197→   197→
   198→   198→  // Setup HLS player (only for HLS URLs, including resolved m3u8)
   199→   199→  useEffect(() => {
   200→   200→    if (!playerVisible || !videoRef.current || !effectiveUrl || !isHls) return;
   201→   201→
   202→   202→    const video = videoRef.current;
   203→   203→    const currentUrl = effectiveUrl;
   204→   204→
   205→   205→    // State auto-resets when URL changes (via derived computation above)
   206→   206→    retryCountRef.current = 0;
   207→   207→
   208→   208→    // Destroy previous HLS instance
   209→   209→    if (hlsRef.current) {
   210→   210→      hlsRef.current.destroy();
   211→   211→      hlsRef.current = null;
   212→   212→    }
   213→   213→
   214→   214→    if (Hls.isSupported()) {
   215→   215→      const hls = new Hls({
   216→   216→        enableWorker: true,
   217→   217→        lowLatencyMode: true,
   218→   218→        maxBufferLength: 30,
   219→   219→        maxMaxBufferLength: 60,
   220→   220→        startLevel: -1,
   221→   221→        manifestLoadingTimeOut: 10000,
   222→   222→        manifestLoadingMaxRetry: 1,
   223→   223→        levelLoadingTimeOut: 10000,
   224→   224→        levelLoadingMaxRetry: 1,

   228→   228→
   229→   229→      hls.loadSource(currentUrl);
   230→   230→      hls.attachMedia(video);
   231→   231→
   232→   232→      hls.on(Hls.Events.MANIFEST_PARSED, () => {

   234→   234→        video.play().then(() => setIsPlaying(true)).catch(() => {});

   237→   237→      hls.on(Hls.Events.ERROR, (_event, data) => {

   240→   240→            case Hls.ErrorTypes.NETWORK_ERROR:
   241→   241→              if (retryCountRef.current < 1) {
   242→   242→                retryCountRef.current += 1;
   243→   243→                hls.startLoad();
   244→   244→              } else {
   245→   245→                if (tryNextChannel()) return;
   246→   246→                setErrorInfo({ url: currentUrl, msg: t(language, 'player.streamUnavailable') });
   247→   247→                hls.destroy();

   249→   249→              break;
   250→   250→            case Hls.ErrorTypes.MEDIA_ERROR:
   251→   251→              hls.recoverMediaError();
   252→   252→              break;
   253→   253→            default:
   254→   254→              if (tryNextChannel()) return;
   255→   255→              setErrorInfo({ url: currentUrl, msg: t(language, 'player.playbackError') });
   256→   256→              hls.destroy();
   257→   257→              break;
   258→   258→          }
   259→   259→        }
   260→   260→      });
   261→   261→
   262→   262→      hlsRef.current = hls;
   263→   263→    } else if (video.canPlayType('application/vnd.apple.mpegurl')) {
   264→   264→      video.src = currentUrl;
   265→   265→      const onLoaded = () => {
   266→   266→        setReadyUrl(currentUrl);
   267→   267→        video.play().then(() => setIsPlaying(true)).catch(() => {});
   268→   268→      };
   269→   269→      const onError = () => {
   270→   270→        if (tryNextChannel()) return;
   271→   271→        setErrorInfo({ url: currentUrl, msg: t(language, 'player.streamUnavailableShort') });
   272→   272→      };
   273→   273→      video.addEventListener('loadedmetadata', onLoaded);
   274→   274→      video.addEventListener('error', onError);
   275→   275→    } else {
   276→   276→      if (!tryNextChannel()) {
   277→   277→        queueMicrotask(() => {
   278→   278→          setErrorInfo({ url: currentUrl, msg: t(language, 'player.hlsNotSupported') });
   279→   279→        });
   280→   280→      }
   281→   281→    }
   282→   282→
   283→   283→    return () => {
   284→   284→      if (hlsRef.current) {
   285→   285→        hlsRef.current.destroy();
   286→   286→        hlsRef.current = null;
   287→   287→      }
   288→   288→    };
   289→   289→  }, [playerVisible, effectiveUrl, isHls, tryNextChannel, language]);
   290→   290→
   291→   291→  // Lock body scroll when player is open
   292→   292→  useEffect(() => {
   293→   293→    if (playerVisible) {
   294→   294→      document.body.style.overflow = 'hidden';
   295→   295→      currentAltIndexRef.current = 0;
   296→   296→    } else {
   297→   297→      document.body.style.overflow = '';
   298→   298→    }
   299→   299→    return () => {
   300→   300→      document.body.style.overflow = '';
   301→   301→    };
   302→   302→  }, [playerVisible]);
   303→   303→
   304→   304→  const togglePlay = () => {
   305→   305→    if (!videoRef.current) return;
   306→   306→    if (isPlaying) {
   307→   307→      videoRef.current.pause();
   308→   308→      setIsPlaying(false);
   309→   309→    } else {
   310→   310→      videoRef.current.play().then(() => setIsPlaying(true)).catch(() => {});
   311→   311→    }
   312→   312→  };
   313→   313→
   314→   314→  const toggleMute = () => {
   315→   315→    if (!videoRef.current) return;
   316→   316→    videoRef.current.muted = !isMuted;
   317→   317→    setIsMuted(!isMuted);
   318→   318→  };
   319→   319→
   320→   320→  const toggleFullscreen = async () => {
   321→   321→    if (!containerRef.current) return;
   322→   322→    if (!isFullscreen) {
   323→   323→      await containerRef.current.requestFullscreen();
   324→   324→      setIsFullscreen(true);
   325→   325→    } else {
   326→   326→      await document.exitFullscreen();
   327→   327→      setIsFullscreen(false);
   328→   328→    }
   329→   329→  };
   330→   330→
   331→   331→  const handleRetry = () => {
   332→   332→    retryCountRef.current = 0;
   333→   333→    setReadyUrl('');
   334→   334→    setErrorInfo(null);
   335→   335→    setResolvedUrl(null);
   336→   336→    setIframeError(false);
   337→   337→    setIframeTimedOut(false);
   338→   338→  };
   339→   339→
   340→   340→  const handleSwitchChannel = (channel: { url: string; name: string; logo?: string }) => {
   341→   341→    retryCountRef.current = 0;
   342→   342→    openPlayer(channel.url, channel.name, channel.logo || undefined);
   343→   343→  };
   344→   344→
   345→   345→  /** Manually try to resolve the current iframe URL to a direct m3u8 stream */
   346→   346→  const handleTryResolve = async () => {
   347→   347→    if (!playerStreamUrl || isResolving) return;
   348→   348→    setIframeError(false);
   349→   349→    setIframeTimedOut(false);
   350→   350→    setIsResolving(true);
   351→   351→    try {
   352→   352→      const res = await fetch('/api/resolve-stream', {
   353→   353→        method: 'POST',
   354→   354→        headers: { 'Content-Type': 'application/json' },
   355→   355→        body: JSON.stringify({ url: playerStreamUrl }),
   356→   356→      });
   357→   357→      const data = await res.json();
   358→   358→      if (data.resolved && data.type === 'hls' && data.resolvedUrl) {
   359→   359→        setResolvedUrl(data.resolvedUrl);
   360→   360→      }
   361→   361→    } catch(_e) {
   362→   362→      // Silently fail
   363→   363→    } finally {
   364→   364→      setIsResolving(false);
   365→   365→    }
   366→   366→  };
   367→   367→
   368→   368→  if (!playerVisible) return null;
   369→   369→
   370→   370→  const hasAlternatives = playerAlternatives && playerAlternatives.length > 0;
   371→   371→
   372→   372→  return (
   373→   373→    <div className="fixed inset-0 z-50 bg-black flex flex-col animate-fade-in">
   374→   374→      {/* Header */}
   375→   375→      <div
   376→   376→        className={`absolute top-0 left-0 right-0 z-10 bg-gradient-to-b from-black/90 via-black/50 to-transparent p-4 pb-8 transition-opacity duration-300 ${showControls || isIframe ? 'opacity-100' : 'opacity-0'}`}
   377→   377→        onMouseMove={hideControlsAfterDelay}
   378→   378→      >
   379→   379→        <div className="flex items-center gap-3">
   380→   380→          <Button
   381→   381→            variant="ghost"
   382→   382→            size="icon"
   383→   383→            onClick={closePlayer}
   384→   384→            className="text-white/90 hover:bg-white/15 h-9 w-9 rounded-xl"
   385→   385→          >
   386→   386→            <X className="h-5 w-5" />
   387→   387→          </Button>
   388→   388→          {playerChannelLogo && (
   389→   389→            <img
   390→   390→              src={playerChannelLogo}
   391→   391→              alt=""
   392→   392→              className="w-8 h-8 rounded-lg object-contain bg-white/10 p-0.5"
   393→   393→              onError={(e) => {
   394→   394→                (e.target as HTMLImageElement).style.display = 'none';
   395→   395→              }}
   396→   396→            />
   397→   397→          )}
   398→   398→          <div className="flex-1 min-w-0">
   399→   399→            <h2 className="text-white font-bold text-sm truncate">{playerChannelName}</h2>
   400→   400→            {isResolving && (
   401→   401→              <span className="text-amber-400/70 text-[10px] flex items-center gap-1">
   402→   402→                <Loader2 className="h-2.5 w-2.5 animate-spin" />
   403→   403→                {t(language, 'player.resolvingStream')}
   404→   404→              </span>
   405→   405→            )}
   406→   406→            {!isResolving && isIframe && (
   407→   407→              <span className="text-emerald-400/60 text-[10px] flex items-center gap-1">
   408→   408→                <Wifi className="h-2.5 w-2.5" />
   409→   409→                {t(language, 'player.live')}
   410→   410→              </span>
   411→   411→            )}
   412→   412→          </div>
   413→   413→          {hasAlternatives && (
   414→   414→            <span className="text-white/30 text-[10px] font-medium">
   415→   415→              +{playerAlternatives.length} {t(language, 'player.otherChannels', playerAlternatives.length)}
   416→   416→            </span>
   417→   417→          )}
   418→   418→        </div>
   419→   419→      </div>
   420→   420→
   421→   421→      {/* Video / Iframe content */}
   422→   422→      <div
   423→   423→        ref={containerRef}
   424→   424→        className="flex-1 flex items-center justify-center relative"
   425→   425→        onMouseMove={hideControlsAfterDelay}
   426→   426→        onClick={isHls ? togglePlay : undefined}
   427→   427→      >
   428→   428→        {isIframe ? (
   429→   429→          /* Iframe-based stream (e.g. from kora-api via proxy) */
   430→   430→          <iframe
   431→   431→            src={iframeSrc}
   432→   432→            className="w-full h-full border-0"
   433→   433→            allow="autoplay; encrypted-media; fullscreen; picture-in-picture"
   434→   434→            allowFullScreen
   435→   435→            title={`${t(language, 'player.liveStream')}: ${playerChannelName}`}
   436→   436→            referrerPolicy="no-referrer"
   437→   437→            onError={() => {
   438→   438→              console.warn('[VideoPlayer] iframe onError triggered');
   439→   439→              setIframeError(true);
   440→   440→            }}
   441→   441→          />

   446→   446→            className="w-full h-full object-contain"

   449→   449→          />
   450→   450→        )}
   451→   451→
   452→   452→        {/* Loading Overlay - only for HLS */}
   453→   453→        {isLoading && (

   456→   456→              <div className="relative">
   457→   457→                <div className="w-16 h-16 rounded-2xl bg-white/5 flex items-center justify-center">
   458→   458→                  <Loader2 className="h-8 w-8 text-white/80 animate-spin" />
   459→   459→                </div>

   462→   462→              <div className="text-center">

   464→   464→                <p className="text-white/30 text-xs mt-1">{t(language, 'player.autoNextChannel')}</p>
   465→   465→              </div>
   466→   466→            </div>
   467→   467→          </div>
   468→   468→        )}
   469→   469→
   470→   470→        {/* Error Overlay (HLS & general errors) */}
